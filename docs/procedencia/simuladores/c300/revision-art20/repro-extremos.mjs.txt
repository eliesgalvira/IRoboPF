import assert from "node:assert/strict"
import { plugin } from "bun"
import { readFileSync } from "node:fs"
const root = "/home/elies/Repos/IRoboPF/"
if (process.argv.includes("--propuesta-original")) {
  const edits = JSON.parse(readFileSync("/home/elies/Backups/IRoboPF/producto-c300-art20-investigacion-20261104/transformaciones.json", "utf8"))
  plugin({ name: "repro-candidata-anterior", setup(b) {
    b.onLoad({ filter: /(reduccion-rendimientos-trabajo|conexion)\.ts$/ }, ({ path }) => {
      const edit = edits.find(e => root + e.path === path)
      return edit ? { contents: edit.after, loader: "ts" } : undefined
    })
  } })
}
const { Effect } = await import(root + "node_modules/effect/dist/index.js")
const { default: Decimal } = await import(root + "node_modules/decimal.js/decimal.mjs")
const { liquidarIrpfAnualExplicito } = await import(root + "lib/dominio/irpf/liquidacion/liquidar-irpf-anual.ts")
const template = JSON.parse(readFileSync(root + "docs/procedencia/implementaciones/redondeo-salarial-open/oficial.json", "utf8"))[0].entrada
let errors = 0
for (const [net, reduction, base] of [["19747.49", "0.01", "17747.48"], ["17673.52", "2364.34", "13309.18"]]) {
  const entry = structuredClone(template)
  const row = entry.trabajo.rendimientos[0]
  row.integro = "900000000000"
  row.cotizaciones.seguridadSocial = "900000000000"
  for (const name of ["sindicato", "colegio", "defensa"]) row[name].importe = "0"
  entry.trabajo.rendimientos = Array.from({ length: 100 }, (_, i) => ({ ...structuredClone(row), id: `r-${i}` }))
  entry.trabajo.rendimientos[0].cotizaciones.seguridadSocial = new Decimal("900000000000").minus(net).toFixed(2)
  entry.retenciones = []
  const result = Effect.runSync(liquidarIrpfAnualExplicito(entry))
  console.log(JSON.stringify({ net, actualReduction: result.reduccionArt20, expectedReduction: reduction, actualBase: result.baseImponibleGeneral, expectedBase: base }))
  try {
    assert(new Decimal(result.reduccionArt20).eq(reduction))
    assert(new Decimal(result.baseImponibleGeneral).eq(base))
  } catch {
    errors++
  }
}
console.log(JSON.stringify({ cases: 2, errors }))
process.exitCode = errors ? 1 : 0
