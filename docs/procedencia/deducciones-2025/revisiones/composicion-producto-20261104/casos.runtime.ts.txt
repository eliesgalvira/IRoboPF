const c = (s: string) => "valenciana_" + s

function peticion(codigo: string, numeros: Record<string, number>, textos: Record<string, string> = {}, cantidad = "g1"): Peticion {
  return { codigo: c(codigo), eleccion: "aplicar", unidades: [{ id: "u", titulares: ["A"], fecha: "2025-06-02", numeros, textos }], referencias: [{ unidad: "u", sujetos: ["persona1"], bienes: ["bien1"], eventos: ["evento1"], cantidades: cantidad ? [cantidad] : [] }] }
}
function entrada(solicitudes: Peticion[], renta = 1000, cantidades = [{ id: "g1", euros: 300 }]): Entrada {
  return { ejercicio: 2025, territorio: "comunitat-valenciana", modalidad: "individual", devengo: "2025-12-31", contribuyente: "A", bases: { liquidableGeneral: renta, liquidableAhorro: 0, imponibleGeneral: renta, imponibleAhorro: 0 }, cantidades, solicitudes, ordenBase30: solicitudes.filter(p => [c(q), c(r), c("conservacion_patrimonio_cultural_titulares"), c("donaciones_cesiones_fines_culturales_cientificos_deportivos")].includes(p.codigo) && p.eleccion === "aplicar").map(p => p.codigo), consignaciones: [] }
}

const total = (a: Entrada, n: number) => { const v = componer(a); assert.equal(v.estado, "calculo_parcial", JSON.stringify(v)); assert.equal(v.totalParcialAntesCuota, n); return v }
