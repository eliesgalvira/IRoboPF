def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def save(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')
    return {'path': str(path), 'sha256': sha(path.read_bytes())}

def page_data(code, n):
    key = (code, n)
    if key not in pages:
        rec = DOCS[code][0]
        with mupdf().open(ORIGINAL / rec['sha256']) as doc:
            p = doc[n - 1]
            box = list(p.rect)
            citation = cite(rec, CAS, n, box, store=False)
            png = crop_bytes(p, box)
            crop = OUT / 'recortes' / (citation['facsimile']['sha256'] + '.png')
            crop.write_bytes(png)
            link = CAS / 'sha256' / citation['facsimile']['sha256']
            if not link.exists():
                link.symlink_to(crop.resolve())
            pages[key] = (citation, p.get_text('words', flags=0, sort=False))
    return pages[key]

def evidence(name, code, page, anchor, literal, occurrence=0, column=None, cell=None, forced_start=None):
    template, words = page_data(code, page)
    citation = json.loads(json.dumps(template))
    text = citation['quote']
    if anchor and anchor not in text:
        raise ValueError((name, repr(anchor), repr(text)))
    lower = text.index(anchor) if anchor else 0
    positions = [m.start() for m in re.finditer(re.escape(literal), text[lower:])]
    start = lower + positions[occurrence] if forced_start is None else forced_start
    unit = next(re.finditer(r'\beuros\b', text[start:], re.I), None)
    if unit is None:
        unit = next(re.finditer(r'\beuros\b', text, re.I))
        ustart = unit.start()
    else:
        ustart = start + unit.start()
    u = text[ustart:ustart + 5]
    # Tablas BOE usan «Euros», no admitido por el lexer: la cabecera «Hasta euros» sí lo es.
    if u not in ['euros', 'EUROS']:
        ustart = text.index('euros')
        u = 'euros'
    binding = {'number': {'start': start, 'end': start + len(literal), 'literal': literal}, 'numberFormat': 'es-ES', 'value': str(Decimal(literal.replace('.', '').replace(',', '.'))), 'unit': {'start': ustart, 'end': ustart + len(u), 'literal': u}, 'unitCode': 'EUR'}
    citation['engineBinding'] = binding
    citation['table'] = {'row': name, 'column': column} if column else None
    validate_schema(citation, 'citation')
    verify(DOCS[code][0], citation, CAS)
    ref = save(OUT / 'citas' / (name + '.json'), citation)
    if cell is None:
        matches = list(re.finditer(r'(?<![0-9.,])' + re.escape(literal) + r'(?![0-9.,])', text))
        ordinal = next(i for i, match in enumerate(matches) if match.start() == start)
        tokens = [w for w in words if w[4] == literal]
        if len(tokens) == len(matches):
            cell = tokens[ordinal][:4]
    citations[name] = {'citation': ref, 'document': DOCS[code][1], 'value': binding['value'], 'cellBBox': cell, 'page': page, 'code': code, 'anchor': anchor, 'occurrence': occurrence}
    return name
