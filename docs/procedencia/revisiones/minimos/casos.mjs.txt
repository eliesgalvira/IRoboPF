import { Effect } from 'effect';
import Decimal from 'decimal.js';
import { writeFileSync } from 'node:fs';
import { obtenerMinimoContribuyente as contrib } from '../../../lib/dominio/irpf/minimos/minimo-contribuyente.ts';
import { obtenerMinimoDescendientes as desc } from '../../../lib/dominio/irpf/minimos/minimo-descendientes.ts';
import { obtenerMinimoAscendientes as asc } from '../../../lib/dominio/irpf/minimos/minimo-ascendientes.ts';
import { obtenerMinimoDiscapacidadFamiliares as disc } from '../../../lib/dominio/irpf/minimos/minimo-discapacidad.ts';
import { obtenerParametrosComunidadAutonoma as params } from '../../../lib/dominio/irpf/comunidades/comunidad-autonoma.ts';
import { liquidarIrpfAnual } from '../../../lib/dominio/irpf/liquidacion/liquidar-irpf-anual.ts';
const none={_tag:'SinDiscapacidad'};
const fam=edad=>({edad,discapacidad:none});
const out={familia:[],annual:[],selectors:[]};
for(const edades of [[25],[30,10],[10],[24],[2],[3],[20,18,16,14,12]]){
 const input=edades.map(fam);out.familia.push({kind:'desc',input,result:String(desc(input))});
}
const disabled={edad:30,discapacidad:{_tag:'Discapacidad33a64',necesitaAyudaOMovilidadReducida:false}};
out.familia.push({kind:'desc-disc-control',input:[disabled,fam(10)],result:String(desc([disabled,fam(10)]))});
for(const edad of [64,65,66,74,75,76]) out.familia.push({kind:'edades',edad,contrib:String(contrib({anio:2025,edad})),asc:String(asc([fam(edad)]))});
for(const discapacidad of [none,{_tag:'Discapacidad33a64',necesitaAyudaOMovilidadReducida:false},{_tag:'Discapacidad33a64',necesitaAyudaOMovilidadReducida:true},{_tag:'Discapacidad65OMas'}])out.familia.push({kind:'discapacidad',input:[{edad:10,discapacidad}],result:String(disc([{edad:10,discapacidad}]))});
out.familia.push({kind:'api-sin-tabla',input:{anio:2012,edad:40},result:String(contrib({anio:2012,edad:40}))});
for(const anio of [2012,2013,2014,2015,2022,2025])for(const comunidadAutonoma of ['madrid','aragon'])for(const gain of [0,400000,1000000]){
 const caso={anio,comunidadAutonoma,situacionFamiliar:{tipo:'individual',edad:40,descendientes:[],ascendientes:[],discapacidad:none},rendimientos:{trabajo:[],gananciasPatrimoniales:[{importeGananciaCentimos:gain,tratamientoMayores65:{_tag:'SinExencionMayores65'},fechaAdquisicion:'2010-01-01',fechaTransmision:anio+'-06-01'}]},reducciones:[],deducciones:[],retencionesSoportadasCentimos:0,pagosACuentaCentimos:0};
 out.annual.push({input:caso,result:Effect.runSync(liquidarIrpfAnual(caso,{modo:'canonico'}))});
}
for(const [anio,comunidadAutonoma,dates] of [[2025,'asturias',['2025-12-02','2025-12-03']],[2023,'illes-balears',['2023-11-25','2023-11-26']],[2022,'comunitat-valenciana',['2022-10-27','2022-10-28']]])for(const date of dates){
 const input={anio,comunidadAutonoma,baseLiquidableGeneral:new Decimal(0),fechaFallecimiento:new Date(date+'T12:00:00Z')};
 out.selectors.push({input,result:params(input)});
}
writeFileSync('docs/procedencia/revisiones/minimos/casos-ejecutados.json',JSON.stringify(out,null,2));
console.log('familia',out.familia,'annual',out.annual.length,'selectors',out.selectors.length);
