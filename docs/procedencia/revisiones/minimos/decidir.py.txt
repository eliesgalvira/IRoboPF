revs = []
for h in hs:
    n = int(h["id"].split("-")[1])
    status, scope, obj, action, refs = dec[n]
    originals = []
    for ref in refs.split(";"):
        if not ref:
            continue
        name, pages = ref.split(":")
        matches = [
            d
            for d in ds
            if (
                d["sha256"].startswith(name[1:])
                if name.startswith("@")
                else d.get("auditName") == name
            )
        ]
        for d in matches:
            originals.append(
                dict(
                    url=d["canonicalUrl"],
                    sha256=d["sha256"],
                    pages=[int(x) for x in pages.split(",") if x],
                    reread=True,
                    scope="Pasajes indicados; no lectura completa del documento ni firma validada",
                )
            )
    revs.append(
        dict(
            id=h["id"],
            previousStatus=h["status"],
            status=status,
            acceptanceScope=scope,
            originalsReread=originals,
            objection=obj,
            reason=scope,
            action=action,
            cost="Aceptación acotada: trabajo de continuidad, condiciones y facsímil permanece pendiente; no corrección automática ni certificación del ejercicio.",
            execution="casos-ejecutados.json, contraste-independiente.json y tablas-runtime.json; los casos de campos ausentes están en casos-no-ejecutados.json",
            engineLocations=h["engineLocations"],
        )
    )
p.joinpath("revisiones.json").write_text(json.dumps(revs, ensure_ascii=False, indent=2))
sources = sorted({loc["path"] for h in hs for loc in h["engineLocations"]})
p.joinpath("huellas-motor.json").write_text(
    json.dumps(
        [
            dict(path=s, sha256=hashlib.sha256(Path(s).read_bytes()).hexdigest())
            for s in sources
        ],
        indent=2,
    )
)
print(
    "decisiones",
    len(revs),
    {
        s: sum(r["status"] == s for r in revs)
        for s in ["accepted", "needs-evidence", "rejected"]
    },
)
