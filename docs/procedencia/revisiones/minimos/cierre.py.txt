p.joinpath("casos-no-ejecutados.json").write_text(
    json.dumps(
        [
            dict(
                id=i,
                inputSpecification=a,
                expectedConditional=b,
                executed=False,
                reason="No representable en contrato actual o no construido como caso anual; no atribuir ejecución a esta especificación.",

scratch = Path(".cache/procedencia/revision-minimos")
manifest = []
for s in scratch.glob("*"):
    if s.suffix not in [".py", ".mjs"]:
        continue
    target = p / (s.name + ".txt")
    target.write_bytes(s.read_bytes())
    manifest.append(
        dict(
            path=str(target),
            sha256=hashlib.sha256(target.read_bytes()).hexdigest(),
            purpose="Receta auxiliar preservada como texto; copiar a scratch para ejecutar, no fuente de aplicación",
        )
    )
p.joinpath("recetas.json").write_text(json.dumps(manifest, indent=2))
rs = json.load(open(p / "revisiones.json"))
tables = json.load(open(p / "tablas-runtime.json"))["tables"]


def leaves(x):
    if isinstance(x, dict):
        return sum(leaves(v) for v in x.values())
    return 1


p.joinpath("cobertura.json").write_text(
