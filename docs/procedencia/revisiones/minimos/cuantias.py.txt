def table(c, d, a, s, sd=None):

            zip(["grado33Hasta65", "grado65OMas", "gastosAsistencia"], map(str, v))

        contribuyente=dict(
            zip(["general", "adicionalMayor65", "adicionalMayor75"], map(str, c))

            zip(

            zip(["mayor65OConDiscapacidad", "adicionalMayor75"], map(str, a))
        ),
        discapacidad=dict(
            contribuyente=disc(s), ascendiente=disc(s), descendiente=disc(sd or s)
        ),
    )
