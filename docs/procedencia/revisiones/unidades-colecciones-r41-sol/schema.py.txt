import gzip,json
from pathlib import Path
from jsonschema import Draft202012Validator
R=Path.cwd(); O=R/'docs/procedencia/revisiones/unidades-colecciones-r41-sol'
results=[]
for name in ['ss','rentas']:
 schema=json.loads((R/f'schemas/procedencia/flujo/{name}-r36.schema.json').read_text());Draft202012Validator.check_schema(schema)
 g=json.loads(gzip.decompress((R/f'docs/procedencia/correcciones/unidades-colecciones-r41/{name}-despues.json.gz').read_bytes()));Draft202012Validator(schema).validate(g)
 results.append({'name':name,'schema':'correcto','candidate':'valido'})
(O/'schemas.json').write_text(json.dumps(results,indent=2));print(results)
