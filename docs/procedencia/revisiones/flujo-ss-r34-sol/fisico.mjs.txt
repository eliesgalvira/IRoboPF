import fs from 'node:fs'
import cp from 'node:child_process'
import crypto from 'node:crypto'
import zlib from 'node:zlib'
const hash=b=>crypto.createHash('sha256').update(b).digest('hex')
const base='d70af62d19291c83d0deb281c09df106ee4a3575'
const entries=cp.execFileSync('git',['ls-tree','-rz','--full-tree',base],{maxBuffer:20e6}).toString().split('\0').filter(Boolean)
const batch=cp.spawn('git',['cat-file','--batch']), records=[]
let buffer=Buffer.alloc(0), pending
batch.stdout.on('data',data=>{buffer=Buffer.concat([buffer,data]);if(pending)pending()})
async function blob(oid){
 batch.stdin.write(oid+'\n')
 return await new Promise(resolve=>{
  pending=()=>{const end=buffer.indexOf(10);if(end<0)return;const size=Number(buffer.subarray(0,end).toString().split(' ')[2]);if(buffer.length<end+size+2)return;const result=Buffer.from(buffer.subarray(end+1,end+size+1));buffer=buffer.subarray(end+size+2);pending=null;resolve(result)};pending()
 })
}
for(const entry of entries){
 const [meta,path]=entry.split('\t'),[mode,,oid]=meta.split(' ')
 const b=await blob(oid)
 try{const s=fs.lstatSync(path),actual=s.isSymbolicLink()?Buffer.from(fs.readlinkSync(path)):fs.readFileSync(path)
 records.push({path,mode,size:actual.length,sha256:hash(actual),baseSha256:hash(b),equal:actual.equals(b),modeEqual:mode==='120000'?s.isSymbolicLink():s.isFile()&&!!(s.mode&0o111)===(mode==='100755')})
 }catch(e){records.push({path,error:String(e),equal:false,modeEqual:false})}
}
batch.stdin.end()
const walk=(p='')=>fs.readdirSync(p||'.',{withFileTypes:true}).flatMap(d=>{const n=p?p+'/'+d.name:d.name;return n==='.git'?[]:d.isDirectory()?walk(n):[n]})
const tracked=new Set(records.map(r=>r.path)),extra=walk().filter(p=>!tracked.has(p))
const badExtra=extra.filter(p=>!p.startsWith('.cache/revision-flujo-ss-r34/')&&!p.startsWith('docs/procedencia/revisiones/flujo-ss-r34-sol/'))
const result={base,head:cp.execFileSync('git',['rev-parse','HEAD']).toString().trim(),entries:records.length,mismatches:records.filter(r=>!r.equal||!r.modeEqual),deletions:records.filter(r=>r.error),badExtra,extra,records,diff:cp.execFileSync('git',['diff',base,'--binary'],{maxBuffer:20e6}).toString()}
fs.writeFileSync('docs/procedencia/revisiones/flujo-ss-r34-sol/fisico-final.json.gz',zlib.gzipSync(JSON.stringify(result)))
console.log(JSON.stringify({...result,records:undefined,extra:extra.length}))
if(result.head!==base||result.entries!==17248||result.mismatches.length||badExtra.length||result.diff)process.exitCode=1
