import fs from 'node:fs';import {spawnSync} from 'node:child_process';import {resolve} from 'node:path';
const root=fs.realpathSync('.'),folder=resolve(root,'docs/procedencia');const names=['inventario.json','inventario-datos.json.gz'];const originals=names.map(n=>fs.readFileSync(resolve(folder,n)));
try{for(const n of names)fs.copyFileSync(resolve(root,'../results/valor/docs/procedencia',n),resolve(folder,n));const r=spawnSync('uv',['run','--python','python3','--no-python-downloads','--with','jsonschema==4.25.1','python','-O','scripts/procedencia/inventario-validar.py'],{encoding:'utf8',env:{...process.env,PYTHONDONTWRITEBYTECODE:'1'}});console.log(JSON.stringify({exit:r.status,stdout:r.stdout,stderr:r.stderr}));}finally{names.forEach((n,i)=>fs.writeFileSync(resolve(folder,n),originals[i]));}
