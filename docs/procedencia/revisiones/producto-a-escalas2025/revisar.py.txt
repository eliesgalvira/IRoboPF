"""Revisión independiente; sólo escribe en esta entrega, no reejecuta generadores del autor."""
from pathlib import Path
import json, hashlib, re, html
from decimal import Decimal as D, ROUND_HALF_UP
import fitz
P=Path(__file__).parent
A=P.parent/'producto-a-escalas2025-20261104'
B=P.parent/'producto-a-cuotas2025-correccion-20261104'
load=lambda p:json.loads(p.read_text())
save=lambda n,x:(P/n).write_text(json.dumps(x,ensure_ascii=False,indent=2)+'\n')
num=lambda s:D(re.sub(r'\s','',s).replace('.','').replace(',','.').replace('%','').replace('»',''))
original=load(A/'originales.json')[0]
assert hashlib.sha256(Path(original['resolvedPath']).read_bytes()).hexdigest()==original['sha256']
doc=fitz.open(original['resolvedPath']);fitz.TOOLS.set_small_glyph_heights(True)
rows={}
for c in load(A/'citas-escalas.json'):
    vals=[]
    for cell in c['columns']:
        literal=doc[c['page']-1].get_textbox(fitz.Rect(cell['bbox']))
        assert ' '.join(literal.split())==' '.join(cell['quote'].split())
        vals.append(literal)
    lo,q,w,t=vals
    rows.setdefault(c['territory'],[]).append([str(num(lo)),str(num(q)),str(num(t)/100)])
for scope in load(A/'ambitos-escalas.json').values():
    for c in scope['contexts']:assert doc[c['page']-1].get_textbox(fitz.Rect(c['bbox']))==c['quote']
    text=' '.join(c['quote'] for c in scope['contexts'])
    assert scope['heading'] in text and 'ejercicio 2025' in text
primary=[]
for c in load(A/'primarios-discrepancias.json'):
    data=Path(c['sourcePath']).read_bytes();assert hashlib.sha256(data).hexdigest()==c['sha256']
    assert data.decode()[c['start']:c['end']]==c['literal']
    parsed=[]
    for tr in re.findall(r'<tr>(.*?)</tr>',c['literal'],re.S):
        cells=re.findall(r'<td[^>]*>(.*?)</td>',tr,re.S)
        if not cells:continue
        cells=[html.unescape(re.sub('<[^>]*>','',x)).strip() for x in cells]
        if cells[0]=='–':
            assert c['territory']=='asturias' and not parsed
            cells[0]='0,00'
        parsed.append([str(num(cells[0])),str(num(cells[1])),str(num(cells[3])/100)])
    assert [[D(x) for x in r] for r in parsed]==[[D(x) for x in r] for r in rows[c['territory']]]
    primary.append({'territory':c['territory'],'rows':len(parsed),'sha256':c['sha256'],'notes':re.findall(r'<p class="nota_pie">(.*?)</p>',c['literal'],re.S)})
quota=lambda rr,b: D(0) if b<=0 else next(D(q)+(b-D(lo))*D(t) for lo,q,t in reversed(rr) if b>=D(lo))
cases=[]
for region in ['aragon','asturias','galicia','castilla-y-leon']:
    rr=rows[region]
    bases={D('-1'),D(0),D('0.001'),D('987654.32')}
    for lo,_,_ in rr:
        bases.update(D(lo)+d for d in [D('-.01'),D('-.001'),D(0),D('.001'),D('.01'),D('137.29')])
    for b in sorted(bases):
        q=quota(rr,b)
        cases.append({'territory':region,'base':str(b),'expected':str(q),'cents':str(q.quantize(D('.01'),rounding=ROUND_HALF_UP))})
result=load(A/'resultado.json');assert len(result['results'])==1477
assert sum(not r['ok'] for r in result['results'])==72
old=[]
for r in result['results']:
    if not r['name'].startswith('quota/'):continue
    _,region,b=r['name'].split('/')
    if region not in rows:continue
    expected=quota(rows[region],D(b));assert expected==D(r['expected'])
    if D(r['actual'])!=expected:old.append(r)
assert len(old)==72
assert sum(D(r['actual']).quantize(D('.01'),rounding=ROUND_HALF_UP)!=D(r['expected']).quantize(D('.01'),rounding=ROUND_HALF_UP) for r in old)==12
oldcases=[]
for r in old:
    _,t,b=r['name'].split('/');q=quota(rows[t],D(b))
    oldcases.append({'territory':t,'base':b,'expected':str(q),'cents':str(q.quantize(D('.01'),rounding=ROUND_HALF_UP))})
net=[]
for t in ['aragon','asturias','galicia','castilla-y-leon']:
    for b,m in [('60123.45','22000'),('17707.20','17707.20'),('10000','22000'),('0','5550')]:
        q=max(D(0),quota(rows[t],D(b))-quota(rows[t],min(D(b),D(m))))
        net.append({'territory':t,'base':b,'minimum':m,'expected':str(q)})
save('oraculo-fronteras.json',{'rows':rows,'cases':cases,'originalDifferenceCases':oldcases,'netCases':net})
save('replay-documental.json',{'physicalRows':sum(map(len,rows.values())),'territories':len(rows),'primaries':primary,'comparisonsReported':1477,'differencesRecalculated':len(old),'centsRecalculated':12,'independentCases':len(cases)})
print('Reextracción física y cuatro primarios correctos;',len(cases),'casos independientes generados')
