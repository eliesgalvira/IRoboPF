const subcase: CasoRetencionTrabajo = { anio: 2020, edad: 40, retribucionAnualCentimos: 1700000, cotizacionesCentimos: 120000, situacionFamiliar: "situacion3", situacionLaboral: "activo", contrato: "general", descendientes: [], ascendientes: [], discapacidad: "sin-discapacidad" }
type Node = { id: string; op: string; args: string[]; data: { key?: string; name?: string }; source: { file: string; start: number; end: number; sha256: string; offsetEncoding: string } }
function failNodes(program: Node[]) {
 const nodes = new Map(program.map(n => [n.id, n]))
 return program.filter(n => {
  const member = nodes.get(n.args[0] ?? "")
  const owner = nodes.get(member?.args[0] ?? "")
  return n.op === "call" && member?.op === "get" && member.data.key === "fail" && owner?.op === "global" && owner.data.name === "Effect"
 })
}
it("B1 transformación exacta, fuente y seis nodos auxiliares por llamada", () => {
 const original = structuredClone(candidate.fragments.retenciones)
 const adapted = preserveFailureOperands(original)
 const fails = failNodes(original.program)
 expect(fails.length).toBe(7)
 const ids = new Set(fails.map(n => n.id))
 for (const before of original.program) {
  const after = adapted.program.find((n: Node) => n.id === before.id)
  if (ids.has(before.id)) {
   expect(after).toEqual({ ...before, op: "throw", args: [before.id+"::r45:signal"] })
   const payload = adapted.program.find((n: Node) => n.id === before.id+"::r45:field:payload")
   expect(payload.args).toEqual([before.args[1]])
   for (const extra of adapted.program.filter((n: Node) => n.id.startsWith(before.id+"::r45:"))) expect(extra.source).toEqual(before.source)
   const source = readFileSync(before.source.file,"utf8").slice(before.source.start,before.source.end)
   expect(source).toContain("Effect.fail(")
  } else expect(after).toEqual(before)
 }
 expect(adapted.program.length-original.program.length).toBe(fails.length*6)
 expect(original).toEqual(candidate.fragments.retenciones)
 expect(() => verifyComposition({ ...candidate, fragments: { ...candidate.fragments, retenciones: adapted } })).toThrow()
})
it("B1 solo patrón exacto, aridad y colisión; laboratorio sobre copia", () => {
 const call = failNodes(candidate.fragments.retenciones.program)[0]!
 for (const args of [[call.args[0]!], [...call.args,call.args[1]!]]) {
  const copy=structuredClone(candidate.fragments.retenciones)
  copy.program.find((n: Node) => n.id===call.id).args=args
  expect(() => preserveFailureOperands(copy)).toThrow("Aridad")
 }
 for (const change of ["owner","member","call"] as const) {
  const copy=structuredClone(candidate.fragments.retenciones)
  const n=copy.program.find((n: Node) => n.id===call.id)
  const member=copy.program.find((x: Node)=> x.id===n.args[0])
  const owner=copy.program.find((x: Node)=> x.id===member.args[0])
  if(change==="owner") owner.data.name="Other"
  if(change==="member") member.data.key="other"
  if(change==="call") n.op="other"
  expect(preserveFailureOperands(copy).program.find((x: Node)=>x.id===n.id)).toEqual(n)
 }
 const collision=structuredClone(candidate.fragments.retenciones)
 collision.program.push({...call,id:call.id+"::r45:signal"})
 expect(()=>preserveFailureOperands(collision)).toThrow("Colisión")
})
it("B2 marcador referencial atraviesa evaluación; copia JSON no autentica pin ni señal", () => {
 const adapted=preserveFailureOperands(candidate.fragments.retenciones)
 const input={...subcase,retribucionAnualCentimos:-1}
 const run=()=>executeFinite(adapted,"calcularRetencionTrabajoAeatImpl",[input,{modo:"canonico"}],false).result
 const result=captureFinite("root",run)
 expect(result.status).toBe("fail")
 interface CapturedException { value?: object }
 const caught: CapturedException={}
 try {run()} catch(error) { if(error instanceof Object) caught.value=error }
 expect(captureFinite("again",()=>{throw caught.value}).status).toBe("fail")
 expect(captureFinite("json",()=>{throw JSON.parse(JSON.stringify(caught.value))}).status).toBe("defect")
 expect(captureFinite("zero",()=>0)).toEqual({status:"success",operation:"zero",value:0})
 for (const error of [Error("ResultadoNoSoportado"),{_tag:"ResultadoNoSoportado"},null,undefined,"fail"]) expect(captureFinite("unknown",()=>{throw error}).status).toBe("defect")
})

  expect(result.value).toEqual(real.success)
  if(modo==="canonico"&&name==="antes reforma") expect(result.value.version).toBe("aeat-2018-01-01")
  if(modo==="canonico"&&name==="después reforma") expect(result.value.version).toBe("aeat-2018-07-05")

  if(!node) throw Error("Nodo causante ausente")

  expect(text).toMatch(/resultadoNoSoportadoRetencion\(\s*caso/)
  if(name==="regularizacion") expect(text).toContain("Regularización no implementada")
  if(name==="negativa") expect(text).toBe("Effect.fail(resultadoNoSoportadoRetencion(caso))")

 }
})
