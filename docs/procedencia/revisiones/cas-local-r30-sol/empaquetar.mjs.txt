import fs from 'node:fs';import cp from 'node:child_process';import {resolve,dirname} from 'node:path';import {createHash} from 'node:crypto';import {gzipSync,gunzipSync} from 'node:zlib';import assert from 'node:assert/strict';
const scope='docs/procedencia/revisiones/cas-local-r30-sol',scratch=resolve('.cache/revision-cas-r30');const hash=b=>createHash('sha256').update(b).digest('hex');const excluded=new Set(['patch.diff.gz','manifiesto.json','transporte.json']);const files=fs.readdirSync(scope).filter(f=>!excluded.has(f)).map(f=>scope+'/'+f).sort();let patch='';const bin=[];for(const p of files){if(p.endsWith('.gz')||fs.statSync(p).size===0){bin.push(p);continue;}const r=cp.spawnSync('diff',['-u','--label','/dev/null','--label','b/'+p,'/dev/null',p],{encoding:'utf8',maxBuffer:20000000});if(r.status!==1&&r.status!==0)throw Error(r.stderr);patch+=r.stdout;}
fs.writeFileSync(scope+'/patch.diff.gz',gzipSync(Buffer.from(patch),{level:9}));const manifest={base:'376fb100512ca6a2f975c3c9500a20ee375dd431',head:cp.execFileSync('git',['rev-parse','HEAD']).toString().trim(),files:files.map(path=>{const b=fs.readFileSync(path);return{path,sha256:hash(b),bytes:b.length,baseSha256:null};}),excludes:['manifiesto.json','transporte.json','patch.diff.gz'],allAddedInScope:true};fs.writeFileSync(scope+'/manifiesto.json',JSON.stringify(manifest,null,2)+'\n');const target=resolve(scratch,'transporte-ensayo');fs.mkdirSync(target);const applied=cp.spawnSync('patch',['-p1','--batch'],{cwd:target,input:gunzipSync(fs.readFileSync(scope+'/patch.diff.gz')),maxBuffer:20000000});fs.writeFileSync(scratch+'/patch-aplicacion.log',applied.stdout);assert.equal(applied.status,0,applied.stderr?.toString());for(const p of bin){fs.mkdirSync(dirname(resolve(target,p)),{recursive:true});fs.copyFileSync(p,resolve(target,p));}for(const f of manifest.files){const b=fs.readFileSync(resolve(target,f.path));assert.equal(hash(b),f.sha256);assert.equal(b.length,f.bytes);}const extra=['patch.diff.gz','manifiesto.json'].map(n=>{const p=scope+'/'+n,b=fs.readFileSync(p);return{path:p,sha256:hash(b),bytes:b.length};});fs.writeFileSync(scope+'/transporte.json',JSON.stringify({base:manifest.base,head:manifest.head,textPatch:extra[0],manifest:extra[1],binaryFiles:manifest.files.filter(f=>bin.includes(f.path)),applied:true,checkedFiles:manifest.files.length,zeroDeleted:true,scratch:target,originals:'originales-preservados.json; no se transportan PDF en Git',selfExcluded:'transporte.json'},null,2)+'\n');console.log('Transporte exacto',files.length);
