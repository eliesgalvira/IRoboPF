import fs from 'node:fs';import {resolve} from 'node:path';import {createHash} from 'node:crypto';import {gzipSync} from 'node:zlib';import assert from 'node:assert/strict';
const out=resolve('.cache/revision-cas-r30'),root=resolve(out,'ensayo'),store=resolve(root,'docs/procedencia/implementaciones/contenidos/datos');const result=JSON.parse(fs.readFileSync(out+'/recorrido.json'));const snap=JSON.parse(fs.readFileSync(resolve(store,'objetos',result.result.snapshot)));const hash=(b,a='sha256')=>createHash(a).update(b).digest('hex');const files=[];for(const h of [result.result.snapshot,...new Set(snap.auxiliares)]){const externo=snap.externos.find(e=>e.hash===h);const path=externo?resolve(out,'externo',h):resolve(store,'objetos',h);const b=fs.readFileSync(path);assert.equal(hash(b),h);files.push({path:path.slice(process.cwd().length+1),sha256:h,bytes:b.length,baseSha256:null,sha512:externo?hash(b,'sha512'):null});}fs.writeFileSync(out+'/cierre-nuevo.json.gz',gzipSync(Buffer.from(JSON.stringify({snapshot:result.result.snapshot,files})),{level:9}));
const old=JSON.parse(fs.readFileSync(out+'/cierre-historico.json'));const oldByHash=new Map(old.files.map(f=>[f.sha256,f]));const common=files.filter(f=>oldByHash.has(f.sha256));assert.ok(snap.externos.every(e=>oldByHash.get(e.hash)?.bytes===e.bytes));fs.writeFileSync(out+'/contraste-historico.json',JSON.stringify({historico:old.snapshot,nuevo:result.result.snapshot,files:files.length,common:common.length,newHashes:files.filter(f=>!oldByHash.has(f.sha256)).map(f=>f.sha256),originalGrandeIdentico:true,sameSnapshotRequired:false},null,2));
const originals=JSON.parse(fs.readFileSync(out+'/originales.json'));fs.writeFileSync(out+'/originales-preservados.json',JSON.stringify(originals.map(o=>({...o,ruta:resolve(root,o.ruta).slice(process.cwd().length+1),metadatos:'hashes, tamaño, clase y MIME; sin cabeceras privadas',preservacionFutura:'Copiar por ambos hashes a una edición nueva; no reexportar v5 r2'})),null,2));console.log('Cierre contrastado',files.length,common.length);
