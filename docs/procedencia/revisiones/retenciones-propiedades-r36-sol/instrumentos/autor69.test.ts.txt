const widths = [1245000n, 775000n, 1500000n, 2480000n, 24000000n],
  rates = [19n, 24n, 30n, 37n, 45n, 47n]
function rational(c: bigint) {
  let rest = c,
    total = 0n
  for (let i = 0; i < rates.length; i++) {
    const used = rest < (widths[i] ?? rest) ? rest : (widths[i] ?? rest)
    total += used * rates[i]
    rest -= used
    if (rest === 0n) break
  }
  return D(total.toString()).div(10000).toFixed()
}
