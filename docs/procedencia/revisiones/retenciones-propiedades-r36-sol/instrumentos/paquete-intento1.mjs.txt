import fs from'node:fs';import path from'node:path';import crypto from'node:crypto';import zlib from'node:zlib';import{execFileSync}from'node:child_process';
const root=process.cwd(),scope='docs/procedencia/revisiones/retenciones-propiedades-r36-sol',out=root+'/'+scope,scratch=root+'/.cache/revision-retenciones-r36',restore=scratch+'/transporte';
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');const excluded=['MANIFIESTO.json','ENTREGA.json','patch.diff.gz','binarios.json','transporte.json'];
function files(dir){return fs.readdirSync(dir,{withFileTypes:true}).flatMap(e=>e.isDirectory()?files(dir+'/'+e.name):[dir+'/'+e.name]);}
const rows=[],binary=[];let patch='';
for(const p of files(out).sort()){if(excluded.includes(path.basename(p)))continue;const b=fs.readFileSync(p),rel=path.relative(root,p);rows.push({path:rel,sha256:sha(b),size:b.length,baseSha256:null,executable:false});if(b.includes(0)||p.endsWith('.gz')){binary.push({path:rel,sha256:sha(b),size:b.length,source:rel});continue;}const lines=b.toString().split('\n');const hasNewline=lines.at(-1)==='';if(hasNewline)lines.pop();patch+=`--- /dev/null\n+++ b/${rel}\n@@ -0,0 +1,${lines.length} @@\n`+lines.map(l=>'+'+l+'\n').join('')+(hasNewline?'':'\\ No newline at end of file\n');}
fs.writeFileSync(out+'/MANIFIESTO.json',JSON.stringify({head:'50f9e949544e20d3e4e0bffc79b926d396267319',baseEntries:20227,baseChanges:[],deletions:[],files:rows},null,2)+'\n');
fs.writeFileSync(out+'/patch.diff.gz',zlib.gzipSync(patch,{level:9}));fs.writeFileSync(out+'/binarios.json',JSON.stringify(binary,null,2)+'\n');fs.mkdirSync(restore,{recursive:true});execFileSync('patch',['-p1','--batch','--forward'],{cwd:restore,input:patch});for(const b of binary){const p=restore+'/'+b.path;fs.mkdirSync(path.dirname(p),{recursive:true});fs.copyFileSync(root+'/'+b.path,p);}for(const row of rows){const b=fs.readFileSync(restore+'/'+row.path);if(sha(b)!==row.sha256||b.length!==row.size)throw Error(row.path);}
fs.writeFileSync(out+'/transporte.json',JSON.stringify({method:'patch -p1 --batch --forward en scratch sin checkout; binarios por ruta/hash',verified:rows.length,allExact:true},null,2)+'\n');
const envelope=['MANIFIESTO.json','patch.diff.gz','binarios.json','transporte.json'].map(name=>{const b=fs.readFileSync(out+'/'+name);return{path:scope+'/'+name,sha256:sha(b),size:b.length};});fs.writeFileSync(out+'/ENTREGA.json',JSON.stringify({head:'50f9e949544e20d3e4e0bffc79b926d396267319',envelope,scope,completeRuleAcceptance:false,parentContrastRequired:true},null,2)+'\n');console.log(JSON.stringify({files:rows.length,binary:binary.length,transportExact:true}));
