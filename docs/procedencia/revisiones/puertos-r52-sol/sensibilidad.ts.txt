const same = (a: ReturnType<typeof ssRun>["result"], b: ReturnType<typeof ssRun>["result"]) => JSON.stringify(a) === JSON.stringify(b)
const indexes = new WeakMap<Fragment, Map<string, Node>>()
function requireNode(fragment: Fragment, id: string): Node {
 let index=indexes.get(fragment)
 if(index === undefined){index=new Map(fragment.program.map(n=>[n.id,n]));indexes.set(fragment,index)}
 const node = index.get(id)
 if (node === undefined) throw Error("Operación ausente: " + id)
 return node
}
function endpoint(fragment: Fragment, root: string) {
 const binding = fragment.globalBindings.find(b => b.name === root)
 if (binding === undefined || binding.node === null || binding.source === undefined) throw Error("Proveedor ausente: " + root)
 const node = requireNode(fragment, binding.node)
 return { root, node: node.id, source: binding.source, signature:signature(binding.source), parameters: node.data.parameters ?? [] }
}
function slice(fragment: Fragment, id: string, seen = new Set<string>()): Node[] {
 if (seen.has(id)) return []
 seen.add(id)
 const n = requireNode(fragment, id)
 return [n, ...n.args.flatMap(a => slice(fragment, a, seen))]
}
function targets(fragment: Fragment, id: string): string[] {
 const n = requireNode(fragment, id)
 if (n.op === "global") return [n.data.name!]
 if (n.op === "conditional") return n.args.slice(1).flatMap(a => targets(fragment, a))
 return []
}
function callsFor(fragment: Fragment & { fields?: {id:string;property:string;unit:string;owner:string|null}[]; propertyBindings?: {id:string;property:string;unit:string;receiver:string}[] }, target: string, constant: boolean): PortCall[] {
 return fragment.program.filter(n => constant ? n.op === "global" && n.data.name === target :
  n.op === "boundaryCall" ? n.data.target === target : n.op === "call" && targets(fragment, n.args[0]!).includes(target)
 ).map(n => ({
  node: n.id, source: n.source,
  arguments: (constant ? [] : n.args.slice(n.op === "boundaryCall" ? 0 : 1)).map((operand,index) => ({ index, operand, binding: slice(fragment, operand) })),
  environment: fragment.program.filter(p=>["function","bind"].includes(p.op) && p.source.file === n.source.file && (p.op === "function" ? p.source.start<=n.source.start && p.source.end>=n.source.end : p.source.start<n.source.start)),
  conditions: fragment.program.filter(p => ["if", "conditional", "switch"].includes(p.op) && p.source.file === n.source.file && p.source.start <= n.source.start && p.source.end >= n.source.end).flatMap(p => slice(fragment,p.args[0]!)),
  frames: fragment.program.filter(p => p.op === "function" && p.source.file === n.source.file && p.source.start <= n.source.start && p.source.end >= n.source.end).map(p=>p.id),
  dimensions: (constant ? [] : n.args.slice(n.op === "boundaryCall" ? 0 : 1)).flatMap(id=>slice(fragment,id)).flatMap(op=>{const field=fragment.fields?.find(f=>f.id===op.id);const fieldScale=fragment.propertyBindings?.find(f=>f.id===op.id);return field !== undefined ? [{node:op.id,property:field.property,unit:field.unit,owner:field.owner}] : fieldScale !== undefined ? [{node:op.id,property:fieldScale.property,unit:fieldScale.unit,owner:fieldScale.receiver}]:[]}),
  selector: constant || n.op === "boundaryCall" ? [] : slice(fragment,n.args[0]!),
 }))
}
function dimensionReference(candidate: Composition, group: Group, source: Source, root: string) {
 if(group === "ss")return {registry:"ss.unitContracts",ids:[root]}
 if(group === "retenciones")return {registry:"retenciones.fieldBindings",ids:candidate.fragments.retenciones.fieldBindings.filter((f:{owner:string})=>["CasoRetencionTrabajo","ContextoRetencionTrabajo"].includes(f.owner)).map((f:{id:string})=>f.id)}
 if(group === "rentas")return {registry:"rentas.fields",ids:candidate.fragments.rentas.fields.filter((f:{source:Source})=>f.source.file === source.file && f.source.start >= source.start && f.source.end <= source.end).map((f:{id:string})=>f.id)}
 return {registry:"escalas.dimensionContracts",ids:root.includes("obtenerParametrosComunidadAutonoma")?["escala","minimos"]:root.includes("Escala")?["escala"]:["minimos"]}
}

export function inspectPortContract(contract: PortContract, candidate: Composition) {
 const expected=producePortContract(candidate)
 const obligations: {id:string;ok:boolean}[]=[]
 const add=(id:string,ok:boolean)=>obligations.push({id,ok})
 add("schema:version",contract.version===expected.version)
 add("schema:domain",contract.domain===expected.domain)
 add("schema:pending",same(contract.pending,expected.pending))
 add("coverage:ports",same(contract.ports.map(p=>p.id).sort(),expected.ports.map(p=>p.id).sort()))
 for(const e of expected.ports){
  const p=contract.ports.find(p=>p.id===e.id)
  for(const key of ["consumer","target","kind","provider","operator","arguments","output","variants","projection","calls"] as const) add(e.id+":"+key,same(p?.[key],e[key]))
  for(const call of e.calls){const got=p?.calls.find(c=>c.node===call.node);for(const key of ["source","arguments","conditions","frames","selector","dimensions","environment"] as const)add(e.id+":"+call.node+":"+key,same(got?.[key],call[key]))}
 }
 add("hashes:dependencies",same(contract.dependencies,expected.dependencies))
 return {status:obligations.every(o=>o.ok)?"ports-satisfied":"ports-pending",obligations,pending:expected.pending}
}
export function verifyPortContract(contract: PortContract, candidate: Composition) {
 checkPortSchema(contract)
 const report=inspectPortContract(contract,candidate)
 const failed=report.obligations.filter(o=>!o.ok)
 if(failed.length > 0) throw Error("Obligación de puerto: "+failed.map(o=>o.id).join(", "))
 return structuredClone(contract)
}

type Value = ReturnType<typeof ssRun>["result"]
export interface SnapshotFields { [key:string]: Snapshot }
export type Snapshot = string | number | boolean | null | undefined | Snapshot[] | SnapshotFields
export interface Witness { port: string; node: string; frames: Snapshot[]; arguments: Snapshot[]; result: Snapshot; bindings: {index:number;operand:string}[]; properties: {path:string;representation:string;value:Snapshot}[]; variant: Variant["when"] | null }
export interface Observation { node: string; target: string; frames: Snapshot[]; arguments: Snapshot[]; result: Snapshot }
function property(value: Value, path: string): Value {
 if(path === "return") return value
 return path.split(".").reduce((v,key)=>key.endsWith("[]")?v?.[key.slice(0,-2)]?.map((x:Value)=>x):Array.isArray(v)?v.map(x=>x?.[key]):v?.[key],value)
}

export function createPortComposition(candidate: Composition, supplied = producePortContract(candidate)) {
 const checked=verifyComposition(candidate)
 const contract=verifyPortContract(supplied,checked)
 return routeContract(checked,contract)
}

export function routeContract(candidate: Composition, contract: PortContract, transform: (target: string, result: Value) => Value = (_target, result) => result) {
 const g=candidate.fragments
 const witnesses: Witness[]=[]
 const scaleBoundaries = new Map<string, (...args: Value[])=>Value>()
 const rentBoundaries: { [name:string]: Value } = {}
 let execution = 0
 const observe=(consumer:"rentas"|"escalas")=>(event:Observation)=>{
  const p=contract.ports.find(p=>p.consumer===consumer && p.target===event.target)
  if(p === undefined) return // Infraestructura de observabilidad y enriquecimiento ya representada en R36, fuera del censo R43.
  const call=p.calls.find(c=>c.node===event.node)
  if(call === undefined) throw Error("Llamada sin contrato: "+p.id+" "+event.node)
  const variant=p.variants.find(v=>property(event.result,v.when.property)===v.when.equals)
  const outputs=variant !== undefined ? p.output.filter(o=>variant.outputs.some(s=>s.split(":")[0]===o.path)) : p.output
  witnesses.push({port:p.id,node:event.node,frames:[{transport:++execution},...event.frames],arguments:event.arguments,result:event.result,bindings:call.arguments.map(a=>({index:a.index,operand:a.operand})),variant:variant?.when??null,properties:outputs.map(o=>({...o,value:property(event.result,o.path)}))})
 }
 const rent=(root:string,args:Value[])=>executeRentasFinite(g.rentas,root,args,false,null,[],rentBoundaries,observe("rentas")).result
 const scales=(root:string,args:Value[])=>executeEscalasR36(g.escalas,root,args,false,null,[],scaleBoundaries,observe("escalas")).result
 for(const p of contract.ports){
  const fragment=g[p.provider.group]
  const registered=p.operator === "evaluate-node" ? p.provider.group === "rentas" && p.provider.root === "expression:"+p.provider.node && fragment.program.some((n:Node)=>n.id===p.provider.node && n.op === "call") : fragment.globalBindings.some((b:Global)=>b.name===p.provider.root && b.node===p.provider.node)
  if(!registered)throw Error("Proveedor ausente: "+p.id)
  const run=(xs:Value[])=>{
   const args=p.arguments.map(a=>xs[a.index])
   if(p.operator === "preadmitted-identity")return args[0]
   const expressionBindings=p.operator === "evaluate-node"?p.provider.parameters.map((parameter,index)=>{if(parameter.binding.name === undefined)throw Error("Binding de decodificación no nominal");return [parameter.binding.name,args[index]]}):[]
   const result=p.operator === "evaluate-node"?executeRentasFinite(g.rentas,null,[],false,p.provider.node,expressionBindings,rentBoundaries,observe("rentas")).result:p.provider.group === "ss"?ssRun(g.ss,p.provider.root,args,false).result:p.provider.group === "retenciones"?retentionRun(g.retenciones,p.provider.root,args,false).result:p.provider.group === "rentas"?rent(p.provider.root,args):scales(p.provider.root,args)
   if(p.operator !== "project" && p.operator !== "evaluate-node")return transform(p.target,result)
   const variant=p.variants.find(v=>property(result,v.when.property)===v.when.equals)
   return Object.fromEntries(p.projection.filter(v=>variant === undefined || variant.outputs.some(o=>o.split(":")[0]===v.to)).map(v=>[v.to,property(result,v.from)]))
  }
  const value=p.operator === "global"?executeEscalasR36(g.escalas,null,[],false,p.provider.node).result:(...xs:Value[])=>run(xs)
  if(p.consumer === "escalas")scaleBoundaries.set(p.target,value)
  else rentBoundaries[p.target]=value
 }
 const ss=createSSEvaluator(g.ss)
 return {
  contract: structuredClone(contract),
  get witnesses(){return structuredClone(witnesses)},
  annual(caso:Parameters<typeof ss>[0],contexto:Parameters<typeof ss>[1]){
   assertPortInputs([caso,contexto])
   const ssResult=ss(caso,contexto)
   const rents=consumeGeneralRentas(g.rentas,caso,contexto,rentBoundaries,observe("rentas"))
   const retention=consumeAnnual(g.retenciones,caso,contexto,false)
   // Wrapper histórico: el observador se transmite por los ejecutores de las raíces.
   const quotas=consumeEscalasR36(g.escalas,"annual",caso,contexto,scaleBoundaries,observe("escalas")).lines
   return {lines:[...ssResult.lines,...rents.lines,...retention.lines,...quotas],admissionClosed:false,wiringAllowed:false}
  },
  explicit(entrada:Parameters<typeof consumeExplicitRentas>[1]){assertPortInputs([entrada]);return consumeExplicitRentas(g.rentas,entrada,rentBoundaries,observe("rentas"))},
  activity(entrada:CasoActividad2025){assertPortInputs([entrada]);return rent("lib/dominio/irpf/actividades-economicas/calcular.ts:calcularActividad2025",[entrada])},
  roots:{rent(root:string,args:Value[]){assertPortInputs(args);return rent(root,args)},scales(root:string,args:Value[]){assertPortInputs(args);return scales(root,args)}},
 }
}
