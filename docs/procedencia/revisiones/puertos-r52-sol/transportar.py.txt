sha=lambda b:hashlib.sha256(b).hexdigest()

for folder in [d,s]:
 for f in sorted(folder.rglob('*')):
  if not f.is_file() or t in f.parents or restore in f.parents:continue
  p=str(f.relative_to(r));b=f.read_bytes();assert len(b)<10*1024*1024
  mode='100755' if f.stat().st_mode&0o111 else '100644';entry=dict(path=p,sha256=sha(b),size=len(b),mode=mode,baseHash=None);manifest.append(entry)
  try:
   text=b.decode('utf8');assert '\0' not in text and len(b)>0
   lines=list(difflib.unified_diff([],text.splitlines(keepends=True),fromfile='/dev/null',tofile=p))
   for line in lines:
    if line.endswith('\n'):patch.append(line)
    else:patch.extend([line+'\n','\\ No newline at end of file\n'])
  except (UnicodeDecodeError,AssertionError):
   (objects/entry['sha256']).write_bytes(b);binaries.append(entry)
