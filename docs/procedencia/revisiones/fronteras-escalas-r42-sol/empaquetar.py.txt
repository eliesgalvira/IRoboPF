for p in payload:
 data=p.read_bytes();rel=str(p.relative_to(r));ex=bool(p.stat().st_mode&0o111)

  text=data.decode('utf8')

 except (UnicodeDecodeError,ValueError):
  obj=objects/item['sha256'];obj.write_bytes(data);binary.append(dict(path=rel,sha256=item['sha256'],size=len(data),object=str(obj.relative_to(r))))

  data=p.read_bytes();files.append(dict(path=str(p.relative_to(r)),sha256=sha(data),size=len(data),baseSha256=None))
