import fs from 'node:fs';import cp from 'node:child_process';import {createHash} from 'node:crypto';import {gzipSync} from 'node:zlib';import {resolve} from 'node:path';import assert from 'node:assert/strict';
const root=process.cwd(),out=resolve('docs/procedencia/revisiones/incorporacion-local-r36-sol'),scratch=resolve('.cache/revision-incorporacion-local-r36'),sha=b=>createHash('sha256').update(b).digest('hex');
fs.copyFileSync(scratch+'/empaquetar.mjs',out+'/empaquetar.mjs.txt');
const envelope=new Set(['manifiesto.json','patch.diff.gz','binarios.json','transporte.json']);const files=fs.readdirSync(out).filter(f=>!envelope.has(f)).sort(),rows=[],bins=[];let patch='';
for(const file of files){const p=out+'/'+file,b=fs.readFileSync(p),relative=p.slice(root.length+1);assert.ok(b.length<=1000000||file.endsWith('.gz'),'Comprimir '+file);const row={path:relative,sha256:sha(b),size:b.length,baseSha256:null};rows.push(row);if(file.endsWith('.gz'))bins.push(row);else{const r=cp.spawnSync('diff',['-u','--label','/dev/null','--label','b/'+relative,'/dev/null',p]);assert.equal(r.status,1);patch+=r.stdout.toString()}}
fs.writeFileSync(out+'/manifiesto.json',JSON.stringify({head:'50f9e949544e20d3e4e0bffc79b926d396267319',entries:rows,excludedEnvelopes:[...envelope],baseChanges:[],deletions:[]},null,2)+'\n');fs.writeFileSync(out+'/binarios.json',JSON.stringify(bins,null,2)+'\n');fs.writeFileSync(out+'/patch.diff.gz',gzipSync(Buffer.from(patch),{level:9}));
const dest=scratch+'/transporte';assert.equal(fs.existsSync(dest),false);fs.mkdirSync(dest);const r=cp.spawnSync('patch',['-p1','--batch','--directory',dest],{input:patch});assert.equal(r.status,0,r.stderr.toString()+r.stdout.toString());
for(const row of bins){const f=dest+'/'+row.path;fs.mkdirSync(resolve(f,'..'),{recursive:true});fs.copyFileSync(root+'/'+row.path,f)}
for(const row of rows){const b=fs.readFileSync(dest+'/'+row.path);assert.equal(sha(b),row.sha256);assert.equal(b.length,row.size)}
const result={head:'50f9e949544e20d3e4e0bffc79b926d396267319',files:rows.length,binaries:bins.length,patchSha256:sha(fs.readFileSync(out+'/patch.diff.gz')),manifestSha256:sha(fs.readFileSync(out+'/manifiesto.json')),bytesExact:true,ordinaryDirectory:true,noCheckoutCreated:true};fs.writeFileSync(out+'/transporte.json',JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(result));
