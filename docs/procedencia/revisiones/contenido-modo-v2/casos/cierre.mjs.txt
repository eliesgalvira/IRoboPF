import {readFileSync,writeFileSync,renameSync} from 'node:fs';
import {resolve} from 'node:path';
import assert from 'node:assert/strict';
import {verificar} from '../../scripts/procedencia/contenido-verificar.mjs';
import {leerSnapshot} from '../../lib/procedencia/almacen.ts';
const root=resolve('.cache/revision-contenido-modo-v2/cas');
const id=JSON.parse(readFileSync('lib/procedencia/rastro-generado.json')).actual.versiones.snapshot;
writeFileSync(resolve(root,'snapshots.json'),JSON.stringify([id]));

renameSync(p,tmp);
