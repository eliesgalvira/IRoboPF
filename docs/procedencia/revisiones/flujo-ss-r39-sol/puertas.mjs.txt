import assert from 'node:assert/strict'
import fs from 'node:fs'
import {composeSS,validateSS,inspectSS} from './proyeccion/scripts/procedencia/flujo-ss-r36.mjs'
import {createSSEvaluator} from './proyeccion/scripts/procedencia/flujo-ss-r36-consumidor.mjs'
const g=composeSS(),rows=[]
for(const key of ['admissionClosed','siteClosed']){const h=structuredClone(g);h.sites[0][key]=true;assert.throws(()=>validateSS(h));rows.push({key,rejected:true})}
