function expected(year, mode, cents) {
  const S = new Decimal(cents).div(100), M = new Decimal(maxima[year-2012])
  const B = mode === 'canonico' && year === 2018
    ? Decimal.min(S,'45014.4').mul(7).plus(Decimal.min(S,'45644.4').mul(5)).div(12)
    : Decimal.min(S,M)
  const E = Decimal.max(0,S.minus(B)), ordinary = [B.mul('.314'),B.mul('.0635')]
  const mei = year < 2023 ? [B.mul(0),B.mul(0)] : [['.005','.001'],['.0058','.0012'],['.0067','.0013'],['.0075','.0015']][year-2023].map(r=>B.mul(r))
  const bands = [Decimal.min(E,M.mul('.1')),Decimal.min(Decimal.max(0,E.minus(M.mul('.1'))),M.mul('.4')),Decimal.max(0,E.minus(M.mul('.5')))]
  const weighted = rates => bands.reduce((s,b,i)=>s.plus(b.mul(rates[i])),new Decimal(0))
  let solid = [new Decimal(0),new Decimal(0)]
  if(year >= 2025) {
    if(mode === 'canonico') {
      const rates = year === 2025 ? [['.0077','.0083','.0098'],['.0015','.0017','.0019']] : [['.0096','.0104','.0122'],['.0019','.0021','.0024']]
      solid = rates.map(weighted)
    } else {
      const total = weighted(year === 2025 ? ['.0092','.0100','.0117'] : ['.0115','.0125','.0146'])
      solid = [total.mul(5).div(6),total.div(6)]
    }
  }
  const totals = ordinary.map((v,i)=>v.plus(mei[i]).plus(solid[i]))
  return [B,E,...ordinary,...mei,...solid,...totals,S.plus(totals[0])]
}
