import fs from 'node:fs';import {syncBuiltinESMExports} from 'node:module';import assert from 'node:assert/strict';import {resolve,dirname} from 'node:path';
const {preparar,aplicar,revertir,sha}=await import('../../scripts/procedencia/incorporacion-lote-r32.mjs');const plan=preparar(),tree=resolve('.cache/incorporacion-lote-r32/revision-r34/parcial');for(const f of plan.files){fs.mkdirSync(dirname(tree+'/'+f.path),{recursive:true});fs.writeFileSync(tree+'/'+f.path,Buffer.from(f.before,'base64'))}const original=fs.writeFileSync;let writes=0;fs.writeFileSync=(...args)=>{if(++writes===2)throw Error('EIO inyectado segunda escritura');return original(...args)};syncBuiltinESMExports();let failure;try{aplicar(tree,plan)}catch(e){failure=String(e)}finally{fs.writeFileSync=original;syncBuiltinESMExports()}assert.match(failure,/EIO/);const hashes=plan.files.map(f=>sha(fs.readFileSync(tree+'/'+f.path)));assert.deepEqual(hashes,[plan.files[0].sha256,plan.files[1].baseSha256]);assert.throws(()=>revertir(tree,plan));assert.throws(()=>aplicar(tree,plan));fs.writeFileSync('.cache/revision-incorporacion-r34/parcial.json',JSON.stringify({failure,hashes,applyRejected:true,revertRejected:true,classification:'limite-serial-declarado',automaticRecovery:false},null,2));console.log('Estado mixto conservado; aplicar/revertir rechazan; sin recuperación automática');
