import fs from 'node:fs';import {resolve} from 'node:path';import {spawnSync} from 'node:child_process';import ts from 'typescript';import Decimal from 'decimal.js';
import {readArtifact} from './scripts/procedencia/inventario-artefacto.mjs';
import {obtenerParametrosComunidadAutonoma as select} from './lib/dominio/irpf/comunidades/comunidad-autonoma.ts';
const {files,manifest}=readArtifact();const items=manifest.pages.flatMap(p=>JSON.parse(files.get(p.file)));const results=[];
function leaves(v,p=[],out=[]){if(v instanceof Decimal || typeof v==='number'){out.push([p,String(v)]);return out}if(v===null || v===undefined)return out;for(const k of Object.keys(v))leaves(v[k],[...p,k],out);return out;}
for(const [year,territory,base,death] of [[2025,'madrid'],[2024,'aragon'],[2026,'simulada-estatal'],[2014,'galicia','17707.20'],[2014,'galicia','17707.21'],[2022,'comunitat-valenciana',null,'2022-10-27T00:00:00.000Z'],[2022,'comunitat-valenciana',null,'2022-10-28T00:00:00.000Z']]){const x=select({anio:year,comunidadAutonoma:territory,baseLiquidableGeneral:base?new Decimal(base):undefined,fechaFallecimiento:death?new Date(death):undefined});const scenario=base?`base=${base}`:death?`fallecimiento=${death}`:'sin entradas opcionales';const actual=leaves({escalaEstatalGeneral:x.escalaEstatalGeneral,escalaAutonomica:x.escalaAutonomica.tramos,minimosAutonomicos:x.minimosAutonomicos});let mismatches=[];for(const [path,value] of actual){const key=JSON.stringify([String(year),territory,scenario,...path]);const entry=items.find(i=>i.layer==='selected'&&i.path===key);if(entry?.value!==value)mismatches.push({key,value,stored:entry?.value})}results.push({year,territory,scenario,leaves:actual.length,mismatches});}
for(const file of ['lib/auditoria-graficos.ts','lib/export/auditoria-excel.ts']){const text=fs.readFileSync(file,'utf8'),sf=ts.createSourceFile(file,text,ts.ScriptTarget.Latest,true);let count=0,missing=[];function visit(n){if(ts.isNumericLiteral(n)){count++;if(!items.some(i=>i.layer==='syntactic'&&i.source.file===file&&i.source.start===n.getStart(sf)&&i.expression===n.getText(sf)))missing.push(n.getText(sf))}ts.forEachChild(n,visit)}visit(sf);results.push({file,numericLiterals:count,missing});}
fs.writeFileSync(resolve(fs.realpathSync('.'),'../results/sample.json'),JSON.stringify(results,null,2));console.log(JSON.stringify(results));
