def sha(b): return hashlib.sha256(b).hexdigest()
def git(root,*args): return subprocess.check_output(['git',*args],cwd=root)
def row(root,name):
 b=(root/name).read_bytes()
 return dict(path=name,sha256=sha(b),size=len(b),baseSha256=None)
lots=[]
for name,work,manifest,pin in [('escalas','t3-bcc14648','MANIFIESTO.json','140daf37a932dd57b33271a61a176bd307c914fbf1dc4d4e5545c02a00c75eea'),('rentas','t3-3f15a57c','ENTREGA.json','84ae4a431ee55263ffa14903b745f1e690309c731f73e663f618ad148af74887')]:
 S=Path('/home/elies/.t3/worktrees/IRoboPF')/work
 F='docs/procedencia/implementaciones/flujo-'+name+'-r36-sol'
 raw=(S/F/manifest).read_bytes(); assert sha(raw)==pin
 m=json.loads(raw); assert m['base']==BASE and git(S,'rev-parse','HEAD').decode().strip()==BASE
 assert not git(S,'diff','--name-status',BASE+'..HEAD').strip()
 if name=='escalas':
  rows=m['files']+[row(S,F+'/'+manifest)]
  payload=m['files'][:-3]
  assert [x['path'] for x in m['files'][-3:]]==[F+'/'+n for n in ['entrega.patch.gz','BINARIOS.json','transporte.json']]
  patchname='entrega.patch.gz'; bins='BINARIOS.json'
 else:
  payload=json.loads((S/m['payload']['manifest']).read_text())['files']
  rows=payload+m['metadata']+[row(S,F+'/'+manifest)]
  patchname='patch.diff.gz'; bins='binarios.json'
 assert len(rows)==len({x['path'] for x in rows})
 assert {str(p.relative_to(S)) for p in (S/F).rglob('*') if p.is_file()}=={e['path'] for e in rows if e['path'].startswith(F+'/')}
 for e in rows:
  p=e['path']; assert '..' not in Path(p).parts and not Path(p).is_absolute()
  assert p.startswith(F+'/') or p.startswith('scripts/procedencia/flujo-'+name+'-r36') and p.endswith('.mjs') or p=='schemas/procedencia/flujo/'+name+'-r36.schema.json' or p=='tests/flujo-'+name+'-r36.test.ts'
  st=(S/p).lstat(); b=(S/p).read_bytes()
  assert stat.S_ISREG(st.st_mode) and not st.st_mode&stat.S_IXUSR
  assert sha(b)==e['sha256'] and len(b)==e['size'] and len(b)<10485760 and e['baseSha256'] is None and not (R/p).exists(),p
 lots.append(dict(name=name,source=S,folder=F,rows=rows,payload=payload,patchname=patchname,bins=bins,pin=pin))

reader=subprocess.Popen(['git','cat-file','--batch'],cwd=R,stdin=subprocess.PIPE,stdout=subprocess.PIPE)
count=0
for entry in git(R,'ls-tree','-rz',BASE).split(b'\0'):

 meta,name=entry.split(b'\t'); mode,kind,oid=meta.split(); assert kind==b'blob'
 reader.stdin.write(oid+b'\n');reader.stdin.flush()
 header=reader.stdout.readline().split();expected=reader.stdout.read(int(header[2]));assert reader.stdout.read(1)==b'\n'
 for lot in lots:
  p=lot['source']/name.decode();st=p.lstat()
  if mode==b'120000':assert stat.S_ISLNK(st.st_mode);actual=os.readlink(p).encode()
  else:
   assert stat.S_ISREG(st.st_mode) and bool(st.st_mode&stat.S_IXUSR)==(mode==b'100755');actual=p.read_bytes()
  assert actual==expected,(lot['name'],name)
 count+=1
reader.stdin.close();assert reader.wait()==0 and count==20227
for lot in lots:
 S=lot['source'];F=lot['folder'];trial=C/('transporte-'+lot['name']);trial.mkdir()
 patch=(S/F/lot['patchname']).read_bytes()
 subprocess.run(['patch','--batch','-p1','-d',str(trial)],input=gzip.decompress(patch),stdout=(C/('transporte-'+lot['name']+'.log')).open('wb'),check=True)
 for e in json.loads((S/F/lot['bins']).read_text()):
  assert any(e['path']==x['path'] and e['sha256']==x['sha256'] and e['size']==x['size'] for x in lot['payload'])
  dst=trial/e['path'];dst.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(S/e['path'],dst)
 for e in lot['payload']:assert sha((trial/e['path']).read_bytes())==e['sha256'],e['path']
 assert {str(p.relative_to(trial)) for p in trial.rglob('*') if p.is_file()}=={e['path'] for e in lot['payload']}

for lot in lots:

 receipt=dict(source=str(lot['source']),base=BASE,manifestSha256=lot['pin'],patchSha256=lot['patchSha256'],physicalEntriesChecked=count,changes=[],deletions=[],transportExact=True,files=lot['rows'])
