import assert from 'node:assert/strict';
import fs from 'node:fs';
import {gzipSync} from 'node:zlib';
import {test} from 'vitest';
import {composeCausalExplicitos} from '/home/elies/Repos/IRoboPF/scripts/procedencia/causal-explicitos-r62-contratos.mjs';
import {executeCausalExplicitos} from '/home/elies/Repos/IRoboPF/scripts/procedencia/causal-explicitos-r62.mjs';
import {explicit} from '/home/elies/Repos/IRoboPF/scripts/procedencia/causal-escalas-r60-casos.mjs';
import {real,Effect} from '/home/elies/Repos/IRoboPF/scripts/procedencia/causal-explicitos-r62-oracle.mjs';
import {causalSlice} from '/home/elies/Repos/IRoboPF/scripts/procedencia/evidencia-ejecucion-r54.ts';
const folder=new URL('./',import.meta.url);
const beforeEditorial = value => JSON.parse(JSON.stringify(value,(key,item)=>key==='procedencia'?undefined:item));
test('independiente: estado anterior, retorno entero e independencia entre ejecuciones',()=>{
 const g=composeCausalExplicitos(),c=explicit('16576'),ctx={route:'explicit',modo:'canonico'};
 const original=JSON.stringify(c);
 const a=executeCausalExplicitos(g,c,ctx,'independiente-a');
 const saved=JSON.stringify(a.evidence);
 const events=a.evidence.events,map=new Map(events.map(e=>[e.id,e]));
 const pushes=events.filter(e=>e.operation==='array-push');assert.equal(pushes.length,2);
 const lengths=events.filter(e=>e.operation==='push-length');assert.deepEqual(lengths.map(e=>e.value),[1,2]);
 const expected=Effect.runSync(real(c));
 const returned=events.find(e=>e.operation==='function-return'&&e.value?._tag==='LiquidacionIrpfAnualExplicitaCalculada');assert(returned);
 // Todo el retorno, no solo pasos o nueve líneas filtradas.
 assert.deepEqual(beforeEditorial(returned.value),beforeEditorial(expected));
 for(let i=0;i<2;i++){
  const push=pushes[i];assert.deepEqual(beforeEditorial(push.value),beforeEditorial(expected.rastro.pasos.slice(0,i+1)));
  const prior=map.get(push.edges.find(e=>e.role==='previous-state').to);assert.equal(prior.operation,'array-selection');assert.equal(prior.property,'push');assert.equal(prior.value,i);
  const ancestors=causalSlice(events,push.id);if(i===1)assert(ancestors.includes(pushes[0].id));
  assert(lengths[i].edges.some(e=>e.role==='mutation-state'&&e.to===push.id));
  assert(causalSlice(events,returned.id).includes(push.id));
 }
 const secondPos=events.indexOf(pushes[1]);
 const firstHistory=events.slice(0,secondPos).filter(e=>Array.isArray(e.value)&&e.value.length===1&&JSON.stringify(e.value[0])===JSON.stringify(pushes[0].value[0]));assert(firstHistory.length>0);
 assert(firstHistory.every(e=>e.value.length===1));
 const b=executeCausalExplicitos(g,c,ctx,'independiente-b');
 assert.equal(JSON.stringify(a.evidence),saved);assert.equal(JSON.stringify(c),original);
 assert.deepEqual(JSON.parse(JSON.stringify(b.evidence).replaceAll('independiente-b','independiente-a')),a.evidence);
 const ids=new Set(b.evidence.events.map(e=>e.id));assert(b.evidence.events.every(e=>e.edges.every(edge=>ids.has(edge.to))));
 const roles=Object.fromEntries(['local','property','parameter:0','return','function-return','array-selection'].map(op=>[op,events.filter(e=>e.operation===op&&Array.isArray(e.value)).length]));
 fs.writeFileSync(new URL('independiente.json',folder),JSON.stringify({fullReturn:true,lengths:[1,2],priorSelections:[0,1],snapshots:[1,2],secondMutationDependsOnFirst:true,historyUnchanged:true,inputUnchanged:true,executionIsolation:true,arrayAliasEvents:roles},null,2));
 fs.writeFileSync(new URL('independiente.json.gz',folder),gzipSync(JSON.stringify({caso:c,contexto:ctx,evidence:a.evidence})));
},600000);
