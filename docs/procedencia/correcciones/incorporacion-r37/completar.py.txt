import hashlib
import json
import subprocess
from pathlib import Path

R = Path('/home/elies/Repos/IRoboPF')
S = R/'.cache/orquestador-recepcion/r37'
O = R/'docs/procedencia/correcciones/incorporacion-r37'
BASE = 'd7d2e2e1687ad9778f6f3c6d27950159aa5078e6'
PARTIAL = 'b40b0fa35683cf7cfdc8144ff2b6ef4082f3e623'
MISSING = {'docs/procedencia/revisiones/incorporacion-local-r36-sol/fisico-final.json.gz', 'docs/procedencia/revisiones/incorporacion-local-r36-sol/fisico-inicial.json.gz'}
PREFIXES = ('docs/procedencia/revisiones/incorporacion-local-r36-sol/', 'docs/procedencia/correcciones/incorporacion-r37/')
EXACT = {'docs/procedencia/ESTADO.md', 'docs/procedencia/DECISIONES.md'}


def cmd(*args):
    return subprocess.check_output(args, cwd=R)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def pending(base):
    paths = set(cmd('git', 'diff', '--name-only', '-z', base).decode().split('\0'))
    paths.update(cmd('git', 'ls-files', '--others', '--exclude-standard', '-z').decode().split('\0'))
    return paths - excluded - {''}


assert cmd('jj', '--ignore-working-copy', 'log', '-r', '@-', '--no-graph', '-T', 'commit_id').decode().strip() == PARTIAL
initial = json.loads((S/'seleccion.json').read_text())
committed = set(cmd('git', 'diff', '--name-only', '-z', BASE, PARTIAL).decode().split('\0'))-{''}
assert committed == {e['path'] for e in initial['files']}-MISSING
for e in initial['files']:
    if e['path'] not in MISSING:
        assert sha(cmd('git', 'show', PARTIAL+':'+e['path'])) == e['sha256']
leftovers = json.loads((R/'docs/procedencia/correcciones/recepcion-r36/postintegracion.json').read_text())['historicalLeftoversExcluded']
excluded = {e['path'] for e in leftovers}
for e in leftovers:
    assert sha((R/e['path']).read_bytes()) == e['sha256']
receipt = json.loads((O/'recepcion.json').read_text())
for e in receipt['files']:
    assert sha((R/e['path']).read_bytes()) == e['sha256']
(O/'completar.py.txt').write_bytes(Path(__file__).read_bytes())
selected = []
for name in sorted(pending(BASE)):
    assert name in EXACT or name.startswith(PREFIXES), name
    data = (R/name).read_bytes()
    assert len(data) < 10485760
    selected.append(dict(path=name, sha256=sha(data), size=len(data)))
manifest_path = O/'confirmacion-final.json'
assert not manifest_path.exists()
manifest_path.write_text(json.dumps(dict(base=BASE, partialCommit=PARTIAL, omittedByDefaultSizeLimit=sorted(MISSING), initialCommittedBlobsVerified=len(committed), receivedFilesVerified=len(receipt['files']), selfExcluded=str(manifest_path.relative_to(R)), files=selected), indent=2)+'\n')
data = manifest_path.read_bytes()
selected.append(dict(path=str(manifest_path.relative_to(R)), sha256=sha(data), size=len(data)))
paths = sorted(pending(PARTIAL))
assert not cmd('git', 'diff', '--name-only', '--diff-filter=D', BASE).strip()
subprocess.run(['jj', '--config', 'snapshot.max-new-file-size=10485760', 'commit', '-m', 'docs(procedencia): conservar cotejos completos de la recepción R37', *['root-file:'+json.dumps(p) for p in paths]], cwd=R, check=True)
commit = cmd('jj', '--ignore-working-copy', 'log', '-r', '@-', '--no-graph', '-T', 'commit_id').decode().strip()
assert set(cmd('git', 'diff', '--name-only', '-z', BASE, commit).decode().split('\0'))-{''} == {e['path'] for e in selected}
assert not cmd('git', 'diff', '--name-only', '--diff-filter=D', BASE, commit).strip()
for e in selected:
    assert sha(cmd('git', 'show', commit+':'+e['path'])) == e['sha256']
for e in leftovers:
    assert sha((R/e['path']).read_bytes()) == e['sha256']
assert not pending(commit)
result = dict(commit=commit, parent=PARTIAL, base=BASE, files=len(selected), bytes=sum(e['size'] for e in selected), deletions=0, allHashesVerified=True)
(S/'commit.json').write_text(json.dumps(result, indent=2)+'\n')
print(json.dumps(result), flush=True)
subprocess.run(['jj', 'bookmark', 'set', 'procedencia-documental', '-r', commit], cwd=R, check=True)
subprocess.run(['jj', 'git', 'push', '--remote', 'origin', '--bookmark', 'procedencia-documental'], cwd=R, check=True)
