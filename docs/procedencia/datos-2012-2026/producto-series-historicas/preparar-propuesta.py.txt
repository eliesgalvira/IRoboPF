from pathlib import Path
import difflib,json,hashlib
out=Path(__file__).resolve().parent; root=Path('/home/elies/Repos/IRoboPF')
p='lib/dominio/irpf/cuotas/escalas-publicadas.ts'; before=(root/p).read_text()
addition='''// Candidato A histórico: BOCM-20181228-1 p4; AEAT 2018-2021.
// Sólo cierre ordinario 31/12; requiere revisión independiente antes de integrar.
export const MADRID_2018_2021 = filas([
  ["0", "0", ".09"],
  ["12450.00", "1120.50", ".112"],
  ["17707.20", "1709.31", ".133"],
  ["33007.20", "3744.21", ".179"],
  ["53407.20", "7395.81", ".21"],
])
'''
after=before.replace('export const MADRID_2022 = filas([',addition+'export const MADRID_2022 = filas([')
needle='  if (comunidadAutonoma === "madrid" && anio === 2022) return MADRID_2022'
