    def test_01_patch_and_manifest_exact(self):
        lines = (SRC / 'propuesta.patch').read_text().splitlines(keepends=True)
        files = {}; current = None
        for line in lines:
            if line.startswith('+++ b/'):
                current = line[6:].strip(); files[current] = ''
            elif current and line.startswith('+'):
                files[current] += line[1:]
        self.assertEqual(set(files), {c['path'] for c in manifest['changes']})
        for c in manifest['changes']:
            data = Path(c['artifact']).read_bytes()
            self.assertEqual(files[c['path']].encode(), data)
            self.assertEqual(hashlib.sha256(data).hexdigest(), c['postimage']['sha256'])
            self.assertFalse((ROOT / c['path']).exists())
        for c in manifest['readDependencies']:
            self.assertEqual(hashlib.sha256((ROOT / c['path']).read_bytes()).hexdigest(), c['checkoutSha256'])

    def test_02_exact_official_bytes_and_independent_text(self):
        evidence = []
        for o in registry:
            raw = store.read(o['document'])
            self.assertEqual(hashlib.sha256(raw).hexdigest(), o['document']['sha256'])
            with flujo.pymupdf.open(stream=raw, filetype='pdf') as pdf:
                self.assertIn('para el ejercicio ' + str(o['year']), ' '.join(pdf[0].get_text().split()))
                for name, proofs in [('title',o['title']),('effects',o['effects']),*o['articles'].items()]:
                    for proof in proofs:
                        box = flujo.pymupdf.Rect(proof['bbox'])
                        words = pdf[proof['page']-1].get_text('words',flags=0,sort=True)
                        text = ' '.join(w[4] for w in words if box.contains(flujo.pymupdf.Rect(w[:4])))
                        self.assertEqual(text, proof['literal'])
                        evidence.append(dict(year=o['year'], section=name, **proof))
                full = '\n'.join(pdf[p-1].get_text(sort=True) for p in sorted({s['page'] for s in o['effects']}))
                self.assertIn('día siguiente', full)
                self.assertIn('enero de ' + str(o['year']), full)
                (OUT / ('seleccion-pdf-' + str(o['year']) + '.txt')).write_text('\n'.join(pdf[p-1].get_text(sort=True) for p in sorted({s['page'] for s in o['title']+o['effects']+sum(o['articles'].values(),[])})))
        (OUT / 'evidencia-selectores.json').write_text(json.dumps(evidence,ensure_ascii=False,indent=2)+'\n')

    def test_03_base_geometry_units_and_groups(self):
        for o in registry:
            for c in o['cells']:
                if c['kind'] != 'base': continue
                v, h, r = (flujo.pymupdf.Rect(c[k]['bbox']) for k in ('value','header','row'))
                self.assertEqual(c['row']['literal'],str(c['group']))
                self.assertEqual(c['periodUnit'], 'EUR/month' if c['group']<=7 else 'EUR/day')
                self.assertIn('Euros/mes' if c['group']<=7 else 'Euros/día', c['header']['literal'])
                self.assertIn('mínimas' if c['column']=='minimum' else 'máximas',c['header']['literal'])
                self.assertEqual(c['value']['page'], c['row']['page'])
                self.assertEqual(c['value']['page'], c['header']['page'])
                self.assertLess(h.y1,v.y0)
                self.assertGreater(min(h.x1,v.x1)-max(h.x0,v.x0),0)
                self.assertGreater(min(r.y1,v.y1)-max(r.y0,v.y0),0)

    def test_06_bindings_current_source_and_dimensions(self):
        for l in self.result['links']:
            if l['status']!='match': continue
            b=l['binding']
            if l['id']=='base-1-maximum':
                self.assertEqual(l['factor'],'12')
                literal = re.search(str(l['year']) + r': crearImporteMonetario\("([0-9.]+)"\)',(ROOT/annual.SS).read_text())[1]
                self.assertEqual(annual.Decimal(literal), annual.Decimal(l['engine']))
            else:
                self.assertEqual(annual.Decimal(l['factor']),annual.Decimal('.01'))
                if l['id'].startswith('solidaridad'):
                    path=json.loads(b['path'])
                    self.assertIn(['PropertyAssignment:'+str(l['year']),0],path)
                    self.assertEqual(path[-1],['StringLiteral',int(l['id'].split('-')[1])-1])
                    source = b['source']
                    raw = (ROOT/source['file']).read_bytes()
                    self.assertEqual(hashlib.sha256(raw).hexdigest(), source['sha256'])
                    utf16 = raw.decode().encode('utf-16-le')
                    expression = utf16[source['start']*2:source['end']*2].decode('utf-16-le')
                    self.assertEqual(expression, b['expression'])
                    self.assertEqual(json.loads(expression), l['engine'])
                elif l['id'].startswith('mei'):
                    self.assertEqual(b['censusParameter']['mei'][b['party']],l['engine'])
                else:
                    self.assertEqual(b['sourceSha256'],hashlib.sha256((ROOT/b['file']).read_bytes()).hexdigest())

    def test_07_adversarial_duplicates_and_stale_binding(self):
        rows=self.result['physicalCells']
        with patch.object(annual,'verify_order',side_effect=lambda store,o:[r for r in rows if r['year']==o['year']]):
            changed=copy.deepcopy(census)
            changed['parameters'].append(copy.deepcopy(next(p for p in census['parameters'] if p['year']==2026)))
            with self.assertRaisesRegex(ValueError,'ambiguo'): annual.run(registry,store,changed,ROOT)
            changed=copy.deepcopy(census)
            next(x for x in changed['leaves'] if x['file']==annual.SS and x['path']==['2025'])['sourceSha256']='0'*64
            with self.assertRaisesRegex(ValueError,'otro módulo'): annual.run(registry,store,changed,ROOT)

    def test_08_census_trust_boundary_reproduction(self):
        # Describe a limitation, not a monetary discrepancy in the exact supplied census.
        changed=copy.deepcopy(census)
        target=next(x for x in changed['leaves'] if x['file']==annual.SS and x['path']==['2025'])
        target['value']='0'
        with patch.object(annual,'verify_order',side_effect=lambda store,o:[r for r in self.result['physicalCells'] if r['year']==o['year']]):
            result=annual.run(registry,store,changed,ROOT)
        self.assertEqual(next(l for l in result['links'] if l['year']==2025 and l['id']=='base-1-maximum')['status'],'difference')
        (OUT/'limite-censo.json').write_text(json.dumps({'reproduction':'Alterar sólo leaves[SS,2025].value a 0, mantener SHA del módulo; run no rechaza el censo, pero devuelve difference. No produce match falso con los bytes exactos entregados.','counts':result['counts']},ensure_ascii=False,indent=2)+'\n')

    def test_09_ast_source_sha_mismatch_and_minimal_fix(self):
        changed=copy.deepcopy(census)
        module=next(m for m in changed['modules'] if m['file']==annual.PROFILE)
        item=next(x for x in module['ast']['items'] if x['symbol']=='tiposSolidaridad' and x.get('value')=='0.0077')
        item['source']['sha256']='0'*64
        rows=self.result['physicalCells']
        original=(SRC/'orden_anual.py').read_text()
        needle="            add(r, item['value'], item, Decimal('.01'))"
        replacement="""            if (item['source']['sha256'] != module['sha256']
                    or item['source']['file'] != PROFILE):
                raise ValueError('AST de otro módulo')
"""+needle
        self.assertEqual(original.count(needle),1)
        fixed=original.replace(needle,replacement)
        (OUT/'revision.patch').write_text(''.join(difflib.unified_diff(original.splitlines(keepends=True),fixed.splitlines(keepends=True),fromfile='a/tools/procedencia/datos/lotes/ipc-ss/orden_anual.py',tofile='b/tools/procedencia/datos/lotes/ipc-ss/orden_anual.py')))
        namespace={'__name__':'review_fixed_annual'}
        exec(compile(fixed,'<review-minimal-fix>','exec'),namespace)
        replay=lambda store,o:[r for r in rows if r['year']==o['year']]
        with patch.object(annual,'verify_order',side_effect=replay):
            result=annual.run(registry,store,changed,ROOT)
        self.assertEqual(result['counts'],{'match':32,'pending':6})
        bad=next(l for l in result['links'] if l['id']=='solidaridad-1-empresarial' and l['year']==2025)
        self.assertEqual(bad['binding']['source']['sha256'],'0'*64)
        namespace['verify_order']=replay
        with self.assertRaisesRegex(ValueError,'AST de otro módulo'):
            namespace['run'](registry,store,changed,ROOT)
        namespace['verify_order']=annual.verify_order
        self.assertEqual(namespace['run'](registry,store,census,ROOT),self.result)
        (OUT/'defecto-reproducible.json').write_text(json.dumps({'severity':'no bloqueante: integridad defensiva del censo alternativo; sin dato público incorrecto en entrega exacta','change':'Sólo item AST tiposSolidaridad 2025 empresa[0].source.sha256 = 64 ceros; módulo y valor intactos','originalCounts':result['counts'],'forgedBinding':bad,'minimalFix':'Comparar source.sha256 y source.file con módulo ya verificado; parche probado en memoria; entrega original preservada','fixedRejects':True,'exactCensusOutputUnchanged':True},ensure_ascii=False,indent=2)+'\n')
