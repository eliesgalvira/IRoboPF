# Preimágenes de la clausura de imports locales, sin inventario de valores/celdas.
roots=['lib/dominio/irpf/comunidades/comunidad-autonoma.ts','lib/dominio/irpf/cuotas/escalas-publicadas.ts','lib/dominio/irpf/cuotas/escalas-gravamen.ts','lib/dominio/irpf/minimos/derechos.ts','lib/dominio/normativa/datos/minimos-autonomicos-2025.ts']
seen={}
def pin(f):
    f=f.resolve()
    if str(f) in seen:return
    data=f.read_bytes();seen[str(f)]=hashlib.sha256(data).hexdigest()
    if f.suffix not in ['.ts','.js','.mjs']:return
    for rel in re.findall(r'(?:from\s*|import\s*)[\x22\x27](\.[^\x22\x27]+)[\x22\x27]',data.decode()):
        target=f.parent/rel
        for candidate in [target,target.with_suffix('.ts'),target/'index.ts']:
            if candidate.is_file():pin(candidate);break
        else:raise ValueError((f,rel))
for r in roots:pin(ROOT/r)
for r in ['package.json','bun.lock']:
    if (ROOT/r).is_file():pin(ROOT/r)
dump('preimagenes.json',seen)
