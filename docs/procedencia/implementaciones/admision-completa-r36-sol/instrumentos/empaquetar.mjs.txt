import fs from'node:fs'
import cp from'node:child_process'
import path from'node:path'
import{createHash}from'node:crypto'
import{gzipSync,gunzipSync}from'node:zlib'
const out='docs/procedencia/implementaciones/admision-completa-r36-sol/',scratch='.cache/admision-completa-r36/',base='50f9e949544e20d3e4e0bffc79b926d396267319',sha=b=>createHash('sha256').update(b).digest('hex')
const terminal=new Set(['MANIFIESTO.json','SOBRE.json','patch.diff.gz','binarios.json','TRANSPORTE.json','COTEJO-POST-TRANSPORTE.json'])
const files=[];function walk(dir){for(const name of fs.readdirSync(dir).sort()){const p=dir+'/'+name;if(fs.lstatSync(p).isDirectory())walk(p);else files.push(p)}}walk(out.slice(0,-1));const payload=files.filter(p=>!terminal.has(p.slice(out.length)))
for(const dir of ['scripts/procedencia','tests','schemas/procedencia/flujo'])for(const n of fs.readdirSync(dir))if((dir==='scripts/procedencia'&&/^flujo-admision-r36.*\.mjs$/.test(n))||(dir==='tests'&&/^flujo-admision-r36.*\.test\.ts$/.test(n))||(dir==='schemas/procedencia/flujo'&&/^admision-r36.*\.schema\.json$/.test(n)))payload.push(dir+'/'+n)
payload.sort();const manifest=[],binaries=[],text=[]
for(const p of payload){if(p.startsWith('/')||p.split('/').includes('..'))throw Error('Ruta inválida');const bytes=fs.readFileSync(p);const exists=cp.spawnSync('git',['cat-file','-e',base+':'+p],{stdio:'ignore'}).status===0;if(exists)throw Error('Modificación de base no autorizada: '+p);const mode=fs.lstatSync(p).mode&0o111?'100755':'100644';manifest.push({path:p,sha256:sha(bytes),size:bytes.length,baseSha256:null,mode});const value=bytes.toString('utf8');if(!Buffer.from(value).equals(bytes)||bytes.includes(0)){binaries.push({path:p,sha256:sha(bytes),size:bytes.length,mode});continue}if(value&&!value.endsWith('\n')){const lines=value.split('\n');text.push('diff --git a/'+p+' b/'+p+'\nnew file mode '+mode+'\n--- /dev/null\n+++ b/'+p+'\n@@ -0,0 +1,'+lines.length+' @@\n'+lines.map(l=>'+'+l+'\n').join('')+'\\ No newline at end of file\n')}else{const lines=value?value.slice(0,-1).split('\n'):[];text.push('diff --git a/'+p+' b/'+p+'\nnew file mode '+mode+'\n'+(lines.length?'--- /dev/null\n+++ b/'+p+'\n@@ -0,0 +1,'+lines.length+' @@\n'+lines.map(l=>'+'+l+'\n').join(''):''))}}
const patch=Buffer.from(text.join(''));fs.writeFileSync(scratch+'patch.diff',patch);fs.writeFileSync(out+'patch.diff.gz',gzipSync(patch,{level:9}));fs.writeFileSync(out+'MANIFIESTO.json',JSON.stringify({base,files:manifest},null,2)+'\n');fs.writeFileSync(out+'binarios.json',JSON.stringify(binaries,null,2)+'\n')
// Intérprete de patch de altas: reconstrucción desde un mapa vacío, sin índice ni checkout nuevo.
const restored=new Map(),lines=gunzipSync(fs.readFileSync(out+'patch.diff.gz')).toString().split('\n');let current=null,body=[],noNewline=false,inHunk=false
function finish(){if(current===null)return;let value=body.join('\n')+(body.length&&!noNewline?'\n':'');restored.set(current,Buffer.from(value));current=null;body=[];noNewline=false;inHunk=false}
for(const line of lines){if(line.startsWith('diff --git ')){finish();current=line.split(' b/')[1]}else if(line.startsWith('@@ '))inHunk=true;else if(inHunk&&line.startsWith('+'))body.push(line.slice(1));else if(line.startsWith('+++ b/')){if(line.slice(6)!==current)throw Error('Ruta de patch incoherente')}else if(line==='\\ No newline at end of file')noNewline=true}finish()
for(const binary of binaries)restored.set(binary.path,fs.readFileSync(binary.path))
const target=scratch+'transporte/';fs.mkdirSync(target,{recursive:true});for(const item of manifest){const bytes=restored.get(item.path);if(!bytes||bytes.length!==item.size||sha(bytes)!==item.sha256)throw Error('Transporte distinto: '+item.path);const dest=target+item.path;fs.mkdirSync(path.dirname(dest),{recursive:true});fs.writeFileSync(dest,bytes);fs.chmodSync(dest,item.mode==='100755'?0o755:0o644);if(!fs.readFileSync(dest).equals(fs.readFileSync(item.path)))throw Error('Disco transportado distinto')}
const check=cp.spawnSync('git',['apply','--check','--unsafe-paths','--directory='+scratch+'comprobacion-patch',scratch+'patch.diff'],{encoding:'utf8'});if(check.status!==0)throw Error('git apply --check: '+check.stderr)
fs.writeFileSync(out+'TRANSPORTE.json',JSON.stringify({base,kind:'additions-only',textPatchGzip:sha(fs.readFileSync(out+'patch.diff.gz')),files:manifest.length,binaries:binaries.length,bytes:manifest.reduce((n,x)=>n+x.size,0),virtualEmptyMapRestoration:true,byteForBytePhysicalScratch:true,gitApplyCheckExitCode:check.status,vcsMutation:false,acceptance:false},null,2)+'\n')
const covered=['MANIFIESTO.json','patch.diff.gz','binarios.json','TRANSPORTE.json'];if(fs.existsSync(out+'COTEJO-POST-TRANSPORTE.json'))covered.push('COTEJO-POST-TRANSPORTE.json')
fs.writeFileSync(out+'SOBRE.json',JSON.stringify({base,files:covered.map(name=>{const b=fs.readFileSync(out+name);return {path:out+name,sha256:sha(b),size:b.length}}),manifestSha256:sha(fs.readFileSync(out+'MANIFIESTO.json')),noSelfReference:true,requires:'parent contrast and new independent review',completeRuleAcceptance:false},null,2)+'\n');console.log(JSON.stringify({files:manifest.length,binaries:binaries.length,manifestSha256:sha(fs.readFileSync(out+'MANIFIESTO.json')),transport:'exact'}))
