import fs from 'node:fs';import path from 'node:path';import {createHash} from 'node:crypto';import {gzipSync,gunzipSync} from 'node:zlib';import {spawnSync,execFileSync} from 'node:child_process';
const base='846f25e183ba07414065d4e6bdf9e98311b98a17',out='docs/procedencia/implementaciones/flujo-retenciones-r32-sol/';const sha=b=>createHash('sha256').update(b).digest('hex');
function walk(dir){return fs.readdirSync(dir,{withFileTypes:true}).flatMap(e=>e.isDirectory()?walk(dir+e.name+'/'):[dir+e.name]);}
const own=[...fs.readdirSync('scripts/procedencia').filter(f=>/^flujo-retenciones-r32.*\.mjs$/.test(f)).map(f=>'scripts/procedencia/'+f),...fs.readdirSync('schemas/procedencia/flujo').filter(f=>/^retenciones-r32.*\.schema\.json$/.test(f)).map(f=>'schemas/procedencia/flujo/'+f),...fs.readdirSync('tests').filter(f=>/^flujo-retenciones-r32.*\.test\.ts$/.test(f)).map(f=>'tests/'+f),...walk(out)].filter(f=>!['MANIFIESTO.json','ENTREGA.json','patch.diff.gz','binarios.json','transporte.json'].some(x=>f===out+x)).sort();
const rows=[],binaries=[];let patch='';for(const file of own){const bytes=fs.readFileSync(file),st=fs.lstatSync(file);const prior=spawnSync('git',['cat-file','blob',base+':'+file]);if(prior.status===0)throw Error('Alta inesperadamente existe en base '+file);rows.push({path:file,sha256:sha(bytes),size:bytes.length,baseSha256:null,executable:Boolean(st.mode&0o111)});if(file.endsWith('.gz'))binaries.push({path:file,sha256:sha(bytes),size:bytes.length});else{const d=spawnSync('git',['diff','--no-index','--','/dev/null',file],{maxBuffer:16*1024*1024});if(d.status!==1)throw Error('Diff falló '+file);patch+=d.stdout.toString();}}
fs.writeFileSync(out+'patch.diff.gz',gzipSync(patch,{level:9}));fs.writeFileSync(out+'binarios.json',JSON.stringify(binaries,null,2)+'\n');fs.writeFileSync(out+'MANIFIESTO.json',JSON.stringify({base,head:execFileSync('git',['rev-parse','HEAD']).toString().trim(),entries:rows,exclusions:['.cache/flujo-retenciones-r32/**','MANIFIESTO.json','ENTREGA.json','patch.diff.gz','binarios.json','transporte.json'],reason:'Sobres fuera de su propio contenido para evitar autorreferencia; ENTREGA fija hashes de todos los sobres.'},null,2)+'\n');
const scratch=fs.mkdtempSync(path.resolve('.cache/flujo-retenciones-r32/transporte-'));const patchPath=path.resolve('.cache/flujo-retenciones-r32/ensayo.diff');fs.writeFileSync(patchPath,gunzipSync(fs.readFileSync(out+'patch.diff.gz')));const result=spawnSync('patch',['-p1','--batch','--directory',scratch,'--input',patchPath],{maxBuffer:8*1024*1024});if(result.status!==0)throw Error(result.stderr+'\n'+result.stdout);for(const b of binaries){const target=path.join(scratch,b.path);fs.mkdirSync(path.dirname(target),{recursive:true});fs.copyFileSync(b.path,target);}for(const r of rows)if(sha(fs.readFileSync(path.join(scratch,r.path)))!==r.sha256)throw Error('Transporte '+r.path);
fs.writeFileSync(out+'transporte.json',JSON.stringify({base,entries:rows.length,bytes:rows.reduce((s,r)=>s+r.size,0),textPatchApplied:true,binariesByPathAndHash:true,allHashesMatched:true,scope:'scratch explícito; no checkout VCS; no aceptación fiscal'},null,2)+'\n');const envelopes=['MANIFIESTO.json','patch.diff.gz','binarios.json','transporte.json'].map(name=>{const b=fs.readFileSync(out+name);return {path:out+name,sha256:sha(b),size:b.length,baseSha256:null};});fs.writeFileSync(out+'ENTREGA.json',JSON.stringify({base,envelopes,completeRuleAcceptance:false,wiringAllowed:false},null,2)+'\n');console.log(JSON.stringify({entries:rows.length,envelopes}));
