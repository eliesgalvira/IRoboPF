import fs from 'node:fs'
const root=process.cwd(),dest=root+'/.cache/flujo-ss-r32/quality'
fs.mkdirSync(dest,{recursive:true})
for(const name of ['lib','types'])if(fs.existsSync(root+'/'+name)&&!fs.existsSync(dest+'/'+name))fs.symlinkSync(root+'/'+name,dest+'/'+name)
for(const n of ['scripts/procedencia','tests'])fs.mkdirSync(dest+'/'+n,{recursive:true})
for(const p of ['scripts/procedencia/flujo-ss-r32.mjs','tests/flujo-ss-r32.test.ts'])fs.copyFileSync(root+'/'+p,dest+'/'+p)
if(!fs.existsSync(dest+'/scripts/procedencia/flujo-ast.mjs'))fs.symlinkSync(root+'/scripts/procedencia/flujo-ast.mjs',dest+'/scripts/procedencia/flujo-ast.mjs')
fs.mkdirSync(dest+'/tools',{recursive:true})
fs.cpSync(root+'/tools/oxlint',dest+'/tools/oxlint',{recursive:true})
fs.copyFileSync(root+'/.oxlintrc.json',dest+'/.oxlintrc.json')
fs.copyFileSync(root+'/eslint.config.mjs',dest+'/eslint.config.mjs')
fs.copyFileSync(root+'/package.json',dest+'/package.json')
if(!fs.existsSync(dest+'/node_modules'))fs.symlinkSync('/home/elies/Repos/IRoboPF/node_modules',dest+'/node_modules')
const config=JSON.parse(fs.readFileSync(root+'/tsconfig.json'))
config.include=['tests/flujo-ss-r32.test.ts','scripts/procedencia/flujo-ss-r32.mjs'];config.compilerOptions.incremental=false
fs.writeFileSync(dest+'/tsconfig.json',JSON.stringify(config,null,2))
const lint=JSON.parse(fs.readFileSync(dest+'/.oxlintrc.json'));if(Object.keys(lint.rules).length!==16||Object.values(lint.rules).some(v=>v!=='error'))throw Error('anti-slop')
