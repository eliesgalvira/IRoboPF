import subprocess,os,stat,hashlib,json,gzip
from pathlib import Path
rows=[]
for entry in subprocess.check_output(['git','ls-tree','-rz','HEAD']).split(b'\0'):
 if not entry: continue
 meta,path=entry.split(b'\t'); mode,kind,oid=meta.split(); p=Path(os.fsdecode(path)); expected=subprocess.check_output(['git','cat-file','blob',oid.decode()]); s=p.lstat(); actual=os.fsencode(os.readlink(p)) if stat.S_ISLNK(s.st_mode) else p.read_bytes()
 rows.append(dict(path=str(p),baseSha256=hashlib.sha256(expected).hexdigest(),sha256=hashlib.sha256(actual).hexdigest(),size=len(actual),mode=mode.decode(),equal=actual==expected and (stat.S_ISLNK(s.st_mode) if mode==b'120000' else stat.S_ISREG(s.st_mode) and bool(s.st_mode&0o111)==(mode==b'100755'))))
result=dict(head=subprocess.check_output(['git','rev-parse','HEAD']).decode().strip(),count=len(rows),differences=[r for r in rows if not r['equal']],entries=rows,diff=subprocess.check_output(['git','diff','HEAD','--stat']).decode())
with gzip.open('.cache/composicion-r43-sol/recepcion-final.json.gz','wt') as f: json.dump(result,f)
print(result['head'],result['count'],result['differences'],result['diff'])
assert len(rows)==21126 and not result['differences']
