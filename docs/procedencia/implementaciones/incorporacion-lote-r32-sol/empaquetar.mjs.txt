import fs from 'node:fs';import {resolve,dirname} from 'node:path';import {execFileSync,spawnSync} from 'node:child_process';import assert from 'node:assert/strict';import {gzipSync,gunzipSync} from 'node:zlib';import {sha,BASE} from '../../scripts/procedencia/incorporacion-lote-r32.mjs';
const scope='docs/procedencia/implementaciones/incorporacion-lote-r32-sol/',own='.cache/incorporacion-lote-r32/',excluded=new Set([scope+'patch.diff.gz',scope+'binarios.json',scope+'manifiesto.json',scope+'transporte.json']);
function files(p){return fs.readdirSync(p,{withFileTypes:true}).flatMap(e=>e.isDirectory()?files(p+'/'+e.name):[p+'/'+e.name])}
const paths=[...files(scope.slice(0,-1)),...files('scripts/procedencia').filter(p=>/^scripts\/procedencia\/incorporacion-lote-r32.*\.mjs$/.test(p)),...files('tests').filter(p=>/^tests\/incorporacion-lote-r32.*\.test\.ts$/.test(p))].filter(p=>!excluded.has(p)).sort();
let patch='';const binaries=[];for(const p of paths){const b=fs.readFileSync(p);assert.ok(b.length<=10000000,p);if(b.includes(0)||p.endsWith('.gz')||!b.length){binaries.push({path:p,sha256:sha(b),size:b.length,baseSha256:null});continue}const run=spawnSync('git',['diff','--no-index','--','/dev/null',p],{encoding:'utf8',maxBuffer:16000000});assert.ok(run.status===0||run.status===1);patch+=run.stdout}
fs.writeFileSync(scope+'patch.diff.gz',gzipSync(Buffer.from(patch),{level:9}));fs.writeFileSync(scope+'binarios.json',JSON.stringify(binaries,null,2)+'\n');const target=resolve(own+'transporte-entrega');fs.mkdirSync(target,{recursive:true});const result=spawnSync('patch',['-p1','-d',target],{input:gunzipSync(fs.readFileSync(scope+'patch.diff.gz')),encoding:'utf8',maxBuffer:16000000});assert.equal(result.status,0,result.stdout+result.stderr);for(const f of binaries){fs.mkdirSync(dirname(resolve(target,f.path)),{recursive:true});fs.copyFileSync(f.path,resolve(target,f.path))}for(const p of paths)assert.equal(sha(fs.readFileSync(resolve(target,p))),sha(fs.readFileSync(p)),p);fs.writeFileSync(scope+'transporte.json',JSON.stringify({success:true,files:paths.length,binaries:binaries.length,patchSha256:sha(fs.readFileSync(scope+'patch.diff.gz')),base:BASE,excludes:['scratch','sobres autorreferentes'],deletions:[]},null,2)+'\n');
const all=[...paths,scope+'patch.diff.gz',scope+'binarios.json',scope+'transporte.json'];fs.writeFileSync(scope+'manifiesto.json',JSON.stringify({base:BASE,HEAD:execFileSync('git',['rev-parse','HEAD']).toString().trim(),files:all.map(path=>{const b=fs.readFileSync(path);return {path,sha256:sha(b),size:b.length,baseSha256:null}}),deletions:[],selfExcluded:true},null,2)+'\n');console.log('Entrega cotejada',all.length)
