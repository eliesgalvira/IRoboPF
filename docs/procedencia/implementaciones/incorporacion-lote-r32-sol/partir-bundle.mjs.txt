import fs from 'node:fs';import {resolve} from 'node:path';import assert from 'node:assert/strict';import {gzipSync,gunzipSync} from 'node:zlib';import {sha} from '../../scripts/procedencia/incorporacion-lote-r32.mjs';
const out=resolve('docs/procedencia/implementaciones/incorporacion-lote-r32-sol'),scratch=resolve('.cache/incorporacion-lote-r32'),p=out+'/snapshot-objetos.json.gz',bytes=fs.readFileSync(p);fs.writeFileSync(scratch+'/snapshot-objetos-original.json.gz',bytes);const parts=[];for(let start=0;start<bytes.length;start+=4000000){const b=bytes.subarray(start,start+4000000),name='snapshot-objetos.gz.part-'+String(parts.length).padStart(3,'0');fs.writeFileSync(out+'/'+name,b);parts.push({path:name,sha256:sha(b),size:b.length})}assert.deepEqual(Buffer.concat(parts.map(e=>fs.readFileSync(out+'/'+e.path))),bytes);fs.writeFileSync(out+'/snapshot-partes.json',JSON.stringify({logicalArchive:'snapshot-objetos.json.gz',sha256:sha(bytes),size:bytes.length,parts,reassembly:'Concatenar parts en este orden; comprobar SHA256 global antes de gunzip y de decodificar base64.',tested:true,pdfOriginals:0,pngOriginals:0},null,2)+'\n');const closure=JSON.parse(gunzipSync(fs.readFileSync(out+'/snapshot-cierre.json.gz')));for(const e of closure.files)if(e.source.startsWith('snapshot-objetos.json.gz#'))e.source=e.source.replace('snapshot-objetos.json.gz#','snapshot-partes.json#');fs.writeFileSync(out+'/snapshot-cierre.json.gz',gzipSync(Buffer.from(JSON.stringify(closure)),{level:9}));fs.unlinkSync(p);console.log('Partes verificadas',parts.length,'gzip original propio conservado en scratch')
