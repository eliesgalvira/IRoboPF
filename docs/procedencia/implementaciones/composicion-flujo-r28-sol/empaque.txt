import fs from 'node:fs'
import cp from 'node:child_process'
import path from 'node:path'
import z from 'node:zlib'
import {sha} from '../../scripts/procedencia/flujo-ast.mjs'
const out='docs/procedencia/implementaciones/composicion-flujo-r28-sol/',base='5e69c68077ffd18be20457160595adb3f0ff52e4'
function files(dir){return fs.readdirSync(dir,{withFileTypes:true}).flatMap(e=>e.isDirectory()?files(path.join(dir,e.name)):[path.join(dir,e.name)])}
const excluded=new Set(['manifiesto.json','patch.diff.gz','binarios.json','transporte.json'])
const own=[...files(out).filter(p=>!excluded.has(path.basename(p))),...files('schemas/procedencia/flujo'),...fs.readdirSync('scripts/procedencia').filter(p=>/^flujo-.*\.mjs$/.test(p)).map(p=>'scripts/procedencia/'+p),...fs.readdirSync('tests').filter(p=>/^flujo-procedencia.*\.test.ts$/.test(p)).map(p=>'tests/'+p)].sort()
const entries=[],patches=[],binaries=[]
for(const p of own){const bytes=fs.readFileSync(p);const original=cp.spawnSync('git',['show',base+':'+p],{encoding:null,maxBuffer:16*1024*1024});const prior=original.status===0?original.stdout:null;entries.push({path:p,sha256:sha(bytes),size:bytes.length,baseSha256:prior?sha(prior):null});if(bytes.length===0||bytes.includes(0)){binaries.push({path:p,sha256:sha(bytes),size:bytes.length});continue}const diff=cp.spawnSync('git',['diff','--no-index','--','/dev/null',p],{encoding:'utf8',maxBuffer:16*1024*1024});if(prior)throw Error('Se esperaba adición exclusiva: '+p);if(diff.status!==1)throw Error('No se pudo crear patch');patches.push(diff.stdout)}
const patch=Buffer.from(patches.join(''));fs.writeFileSync(out+'patch.diff.gz',z.gzipSync(patch,{level:9}));fs.writeFileSync(out+'binarios.json',JSON.stringify(binaries,null,2)+'\n')
const scratch=path.resolve('.cache/composicion-flujo-r28-sol/transporte');fs.mkdirSync(scratch,{recursive:true});const apply=cp.spawnSync('patch',['-p1','--directory='+scratch],{input:patch,encoding:'utf8'});if(apply.status!==0)throw Error(apply.stderr)
for(const b of binaries){const target=path.join(scratch,b.path);fs.mkdirSync(path.dirname(target),{recursive:true});fs.copyFileSync(b.path,target)}
for(const e of entries){const bytes=fs.readFileSync(path.join(scratch,e.path));if(sha(bytes)!==e.sha256||bytes.length!==e.size)throw Error('Transporte divergente '+e.path)}
fs.writeFileSync(out+'transporte.json',JSON.stringify({base,files:entries.length,transport:'patch textual gzip + binarios por ruta/hash, aplicado solo en scratch propio',exact:true},null,2)+'\n')
fs.writeFileSync(out+'manifiesto.json',JSON.stringify({base,head:cp.execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).trim(),scope:'adiciones exclusivas',files:entries,transportFiles:['patch.diff.gz','binarios.json','transporte.json'].map(name=>({path:out+name,sha256:sha(fs.readFileSync(out+name)),size:fs.statSync(out+name).size})),excludedSelf:'manifiesto.json',completeRuleAcceptance:false,review:'pending-independent'},null,2)+'\n')
console.log(JSON.stringify({files:entries.length,binaries:binaries.length,transport:true}))
