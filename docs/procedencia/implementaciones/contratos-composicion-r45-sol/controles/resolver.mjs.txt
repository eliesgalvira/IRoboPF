import { registerHooks, createRequire } from 'node:module'
import { pathToFileURL } from 'node:url'
const parent = createRequire('/home/elies/Repos/IRoboPF/package.json')
registerHooks({ resolve(specifier, context, next) {
 try { return next(specifier, context) } catch (error) {
  if (specifier.startsWith('.') || specifier.startsWith('/') || specifier.startsWith('node:')) throw error
  return { url: pathToFileURL(parent.resolve(specifier)).href, shortCircuit: true }
 }
}})
