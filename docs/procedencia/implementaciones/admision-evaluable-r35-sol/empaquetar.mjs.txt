import fs from 'node:fs'
import cp from 'node:child_process'
import crypto from 'node:crypto'
import zlib from 'node:zlib'
import path from 'node:path'
const out='docs/procedencia/implementaciones/admision-evaluable-r35-sol/'
const cache='.cache/admision-evaluable-r35/'
const hash=b=>crypto.createHash('sha256').update(b).digest('hex')
const exclusions=new Set(['MANIFIESTO.json','SOBRE.json','TRANSPORTE.json','patch.diff.gz','binarios.json'])
const files=[]
function walk(dir){for(const e of fs.readdirSync(dir,{withFileTypes:true})){const p=dir+e.name;if(e.isDirectory())walk(p+'/');else if(!exclusions.has(e.name))files.push(p)}}
walk(out)
for(const dir of ['scripts/procedencia/','schemas/procedencia/flujo/','tests/'])for(const n of fs.readdirSync(dir))if(n.startsWith(dir==='schemas/procedencia/flujo/'?'admision-r35':'flujo-admision-r35'))files.push(dir+n)
const rows=files.sort().map(p=>{const b=fs.readFileSync(p);return {path:p,sha256:hash(b),size:b.length,baseSha256:null,mode:'100644'}})
fs.writeFileSync(out+'MANIFIESTO.json',JSON.stringify({base:'9c3edec704ead945b17e419d2e3ef0d1e8a75a29',files:rows},null,2)+'\n')
const binaries=rows.filter(r=>r.path.endsWith('.gz'))
const texts=[...rows.filter(r=>!r.path.endsWith('.gz')).map(r=>r.path),out+'MANIFIESTO.json']
let patch=''
for(const p of texts) {try{cp.execFileSync('git',['diff','--no-index','--','/dev/null',p],{maxBuffer:10e6})}catch(e){if(e.status!==1)throw e;patch+=e.stdout.toString()}}
fs.writeFileSync(out+'patch.diff.gz',zlib.gzipSync(patch))
fs.writeFileSync(out+'binarios.json',JSON.stringify(binaries,null,2)+'\n')
const restore=path.resolve(cache+'transporte-'+Date.now())
fs.mkdirSync(restore,{recursive:true})
const applied=cp.spawnSync('patch',['-p1','--batch'],{cwd:restore,input:patch,encoding:'utf8',maxBuffer:1e6})
if(applied.status!==0)throw Error(applied.stderr+applied.stdout)
for(const r of binaries){const target=restore+'/'+r.path;fs.mkdirSync(path.dirname(target),{recursive:true});fs.copyFileSync(r.path,target)}
for(const r of rows){const b=fs.readFileSync(restore+'/'+r.path);if(hash(b)!==r.sha256||b.length!==r.size)throw Error('Transporte: '+r.path)}
fs.writeFileSync(out+'TRANSPORTE.json',JSON.stringify({status:'byte-equal',fileCount:rows.length,restore,method:'patch -p1 --batch sobre scratch vacío; binarios por ruta/hash; cotejo SHA256/tamaño de cada archivo',partialImplementation:true},null,2)+'\n')
const envelope=['MANIFIESTO.json','patch.diff.gz','binarios.json','TRANSPORTE.json'].map(n=>{const b=fs.readFileSync(out+n);return {path:out+n,sha256:hash(b),size:b.length}})
fs.writeFileSync(out+'SOBRE.json',JSON.stringify({files:envelope,completeRuleAcceptance:false,wiringAllowed:false},null,2)+'\n')
console.log({files:rows.length,patchBytes:patch.length,binaries:binaries.length,transport:'byte-equal'})
